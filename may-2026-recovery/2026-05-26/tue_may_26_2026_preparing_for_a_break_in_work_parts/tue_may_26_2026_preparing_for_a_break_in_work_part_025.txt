New-Item -ItemType Directory -Path $partsFolder -Force | Out-Null

Get-ChildItem $partsFolder -ErrorAction SilentlyContinue |

  Remove-Item -Force -ErrorAction SilentlyContinue

$chunk = New-Object System.Collections.Generic.List[string]
$partNumber = 1

Get-Content $SourceFile | ForEach-Object {
    $chunk.Add($_)

    if ($chunk.Count -ge $LinesPerPart) {
        $partPath = Join-Path $partsFolder ("{0}_part_{1:D3}.txt" -f $baseName, $partNumber)
        $chunk | Set-Content $partPath -Encoding UTF8
        $chunk = New-Object System.Collections.Generic.List[string]
        $partNumber++
    }
}

if ($chunk.Count -gt 0) {
    $partPath = Join-Path $partsFolder ("{0}_part_{1:D3}.txt" -f $baseName, $partNumber)
    $chunk | Set-Content $partPath -Encoding UTF8
}

Get-ChildItem $partsFolder |
  Select-Object Name, Length, LastWriteTime

function Write-Info {

function Get-VersionNumber {
    param([string]$FileName)

    if ($FileName -match '\((\d+)\)(?=\.[^.]+$)') {
        return [int]$matches[1]
    }

    return 0
}

function Get-NormalizedBaseName {
    param([string]$FileName)

    $nameWithoutExtension = [System.IO.Path]::GetFileNameWithoutExtension($FileName)
    $normalized = $nameWithoutExtension -replace '\s*\(\d+\)$', ''
    return $normalized.ToLower().Replace("-", "_").Replace(" ", "_")
}

function Get-DatePatterns {
    param([datetime]$Date)

    return @(
        $Date.ToString("ddd_MMM_d_yyyy"),
        $Date.ToString("ddd_MMMM_d_yyyy"),
        $Date.ToString("yyyy-MM-dd"),
        $Date.ToString("MM-dd-yyyy"),
        $Date.ToString("M-d-yyyy")
    ) | ForEach-Object {
        $_.ToLower().Replace("-", "_").Replace(" ", "_")
    }
}

function Get-ExportFilesForDate {
    param(
        [string[]]$Paths,
        [datetime]$Date
    )

    $datePatterns = Get-DatePatterns -Date $Date
    $results = @()

    foreach ($path in $Paths) {
        if (-not (Test-Path $path)) {
            continue
        }

        $files = Get-ChildItem -Path $path -ErrorAction SilentlyContinue |
            Where-Object { -not $_.PSIsContainer } |
            Where-Object { $_.Extension.ToLower() -in ".md", ".json", ".zip" } |
            Where-Object {
                $baseName = $_.BaseName.ToLower().Replace("-", "_").Replace(" ", "_")
                ($datePatterns | Where-Object { $baseName -like "*$_*" }).Count -gt 0
            }

        $results += $files
    }

    return $results
}

function Get-LatestExportGroup {
    param([object[]]$Files)

    if (-not $Files -or $Files.Count -eq 0) {
        return $null
    }

    $ranked = $Files |
        Sort-Object `
            @{ Expression = { Get-VersionNumber $_.Name }; Descending = $true },
            @{ Expression = { $_.LastWriteTime }; Descending = $true }

    $best = $ranked | Select-Object -First 1
    $bestNormalized = Get-NormalizedBaseName $best.Name
    $bestVersion = Get-VersionNumber $best.Name

    return $Files | Where-Object {
        (Get-NormalizedBaseName $_.Name) -eq $bestNormalized -and
        (Get-VersionNumber $_.Name) -eq $bestVersion
    }
}

function Expand-ZipFile {
    param(
        [Parameter(Mandatory = $true)]
        [object]$ZipFile,

        [Parameter(Mandatory = $true)]
        [string]$DestinationRoot
    )

    $extractFolderName = [System.IO.Path]::GetFileNameWithoutExtension($ZipFile.Name)
    $extractPath = Join-Path $DestinationRoot $extractFolderName

    if (Test-Path $extractPath) {
        Remove-Item -Path $extractPath -Recurse -Force -ErrorAction SilentlyContinue
    }

    New-Item -ItemType Directory -Path $extractPath -Force | Out-Null
    Expand-Archive -Path $ZipFile.FullName -DestinationPath $extractPath -Force

    return $extractPath
}

function Get-MainMarkdownCandidate {
    param(
        [object[]]$MarkdownFiles,
        [string]$ExpectedBaseName
    )

    if (-not $MarkdownFiles -or $MarkdownFiles.Count -eq 0) {
        return $null
    }

    $expected = $ExpectedBaseName.ToLower()

    $ranked = $MarkdownFiles |
        Sort-Object `
            @{ Expression = {
                $name = $_.BaseName.ToLower()
                if ($name -eq $expected) { 100 }
                elseif ($name -like "$expected*") { 90 }
                elseif ($name -match 'checklist|readme|guide|instruction|instructions') { 10 }
                else { 50 }
            }; Descending = $true },
            @{ Expression = { $_.Length }; Descending = $true },
            @{ Expression = { $_.LastWriteTime }; Descending = $true }

    return $ranked | Select-Object -First 1
}

function Get-BestImportSourceFromGroup {
    param(
        [object[]]$GroupFiles,
        [string]$TodayFolder
    )

    if (-not $GroupFiles -or $GroupFiles.Count -eq 0) {
        return $null
    }

    $zipFile = $GroupFiles |
        Where-Object { $_.Extension.ToLower() -eq ".zip" } |
        Sort-Object LastWriteTime -Descending |
        Select-Object -First 1

    if ($zipFile) {
        $extractRoot = Join-Path $TodayFolder "extracted"
        New-Item -ItemType Directory -Path $extractRoot -Force | Out-Null

        $extractPath = Expand-ZipFile -ZipFile $zipFile -DestinationRoot $extractRoot
        $expectedBaseName = Get-NormalizedBaseName $zipFile.Name

        $mdFiles = Get-ChildItem -Path $extractPath -Recurse -ErrorAction SilentlyContinue |
            Where-Object { -not $_.PSIsContainer -and $_.Extension.ToLower() -eq ".md" }

        $mdInside = Get-MainMarkdownCandidate -MarkdownFiles $mdFiles -ExpectedBaseName $expectedBaseName
        if ($mdInside) {
            return [PSCustomObject]@{
                SourceType    = "ExtractedMarkdown"
                ImportFile    = $mdInside
                ContainerFile = $zipFile
                ExtractPath   = $extractPath
            }
        }

        $jsonInside = Get-ChildItem -Path $extractPath -Recurse -ErrorAction SilentlyContinue |
            Where-Object { -not $_.PSIsContainer -and $_.Extension.ToLower() -eq ".json" } |
            Sort-Object `
                @{ Expression = { $_.Length }; Descending = $true },
                @{ Expression = { $_.LastWriteTime }; Descending = $true } |
            Select-Object -First 1

        if ($jsonInside) {
            return [PSCustomObject]@{
                SourceType    = "ExtractedJson"
                ImportFile    = $jsonInside
                ContainerFile = $zipFile
                ExtractPath   = $extractPath
            }
        }
    }

    $standaloneMdFiles = $GroupFiles |
        Where-Object { $_.Extension.ToLower() -eq ".md" }

    $expectedStandaloneBase = Get-NormalizedBaseName (($GroupFiles | Select-Object -First 1).Name)
    $standaloneMd = Get-MainMarkdownCandidate -MarkdownFiles $standaloneMdFiles -ExpectedBaseName $expectedStandaloneBase

    if ($standaloneMd) {
        return [PSCustomObject]@{
            SourceType    = "StandaloneMarkdown"
            ImportFile    = $standaloneMd
            ContainerFile = $null
            ExtractPath   = $null
        }
    }

    $standaloneJson = $GroupFiles |
        Where-Object { $_.Extension.ToLower() -eq ".json" } |
        Sort-Object `
            @{ Expression = { $_.Length }; Descending = $true },
            @{ Expression = { $_.LastWriteTime }; Descending = $true } |
        Select-Object -First 1

    if ($standaloneJson) {
        return [PSCustomObject]@{
            SourceType    = "StandaloneJson"
            ImportFile    = $standaloneJson
            ContainerFile = $null
            ExtractPath   = $null
        }
    }

    return $null
}

function Get-LatestPreviousHistoryFile {
    param(
        [string]$BasePath,
        [string]$Today
    )

    $previousFolder = Get-ChildItem -Path $BasePath -ErrorAction SilentlyContinue |
        Where-Object { $_.PSIsContainer -and $_.Name -lt $Today } |
        Sort-Object Name -Descending |
        Select-Object -First 1

    if (-not $previousFolder) {
        return $null
    }

    return Get-ChildItem -Path $previousFolder.FullName -ErrorAction SilentlyContinue |
        Where-Object { -not $_.PSIsContainer -and $_.Name -like "*_master-history.md" } |
        Sort-Object LastWriteTime -Descending |
        Select-Object -First 1
}

function Build-MasterHistoryFile {
    param(
        [datetime]$Date,
        [string]$DailyBasePath,
        [object]$ImportSource
    )

    $today = $Date.ToString("yyyy-MM-dd")
    $todayFolder = Join-Path $DailyBasePath $today
    New-Item -ItemType Directory -Path $todayFolder -Force | Out-Null

    $historyFile = Join-Path $todayFolder "${today}_master-history.md"
    $previousHistory = Get-LatestPreviousHistoryFile -BasePath $DailyBasePath -Today $today
    $latestImportSource = if ($ImportSource) { $ImportSource.ImportFile.FullName } else { "None" }

    $content = @()
    $content += "# Daily Copilot Running History"
    $content += ""
    $content += "- Date: $today"
    $content += "- Last Updated: $((Get-Date).ToString('HH:mm:ss'))"
    $content += "- Latest Import Source: $latestImportSource"
    $content += ""

    if ($ImportSource) {
        $content += "- Import Source Type: $($ImportSource.SourceType)"
        if ($ImportSource.ContainerFile) {
            $content += "- Source Archive: $($ImportSource.ContainerFile.FullName)"
        }
        if ($ImportSource.ExtractPath) {
            $content += "- Extracted To: $($ImportSource.ExtractPath)"
        }
        $content += ""
    }

    $content += "---"
    $content += ""
    $content += "## Previous Running History"
    $content += ""

    if ($previousHistory) {
        $content += Get-Content -Path $previousHistory.FullName
    }
    else {
        $content += "_No previous running history file was found._"
    }

    $content += ""
    $content += "---"
    $content += ""
    $content += "## Today's Latest Export"
    $content += ""

    if ($ImportSource) {
        $detectedVersionSource = if ($ImportSource.ContainerFile) { $ImportSource.ContainerFile.Name } else { $ImportSource.ImportFile.Name }

        $content += "- Source File: $($ImportSource.ImportFile.FullName)"
        $content += "- Last Write Time: $($ImportSource.ImportFile.LastWriteTime)"
        $content += "- File Type: $($ImportSource.ImportFile.Extension)"
        $content += "- Detected Version: $(Get-VersionNumber $detectedVersionSource)"
        $content += ""
        $content += "===== BEGIN IMPORTED EXPORT ====="
        $content += ""

        try {
            $raw = Get-Content -Path $ImportSource.ImportFile.FullName -Raw
            $content += $raw
        }
        catch {
            $content += "[ERROR] Failed to read import file: $($ImportSource.ImportFile.FullName)"
            $content += $_.Exception.Message
        }

        $content += ""
        $content += "===== END IMPORTED EXPORT ====="
    }
    else {
        $content += "_No export file for target date was found in the scan paths._"
    }

    $content += ""
    $content += "---"
    $content += ""
    $content += "## New Notes"
    $content += ""
    $content += "_Add today's notes below._"
    $content += ""

    Set-Content -Path $historyFile -Value $content -Encoding UTF8
    return $historyFile
}

function Build-ExecutiveSummaryFile {
    param(
        [datetime]$Date,
        [string]$TargetFolder,
        [string]$MasterHistoryPath,
        [object]$ImportSource
    )

    $dateString = $Date.ToString("yyyy-MM-dd")
    $summaryPath = Join-Path $TargetFolder "${dateString}_executive-summary.md"

    $summary = @()
    $summary += "# Executive Summary - $dateString"
    $summary += ""
    $summary += "## Source Overview"
    $summary += ""
    $summary += "- Master History: $MasterHistoryPath"

    if ($ImportSource) {
        $summary += "- Import Source Type: $($ImportSource.SourceType)"
        $summary += "- Import File: $($ImportSource.ImportFile.FullName)"
        if ($ImportSource.ContainerFile) {
            $summary += "- Archive: $($ImportSource.ContainerFile.FullName)"
        }
        if ($ImportSource.ExtractPath) {
            $summary += "- Extracted Folder: $($ImportSource.ExtractPath)"
        }
    }
    else {
        $summary += "- No import source detected for the target date."
    }

    $summary += ""
    $summary += "## Notes"
    $summary += ""
    $summary += "- This file is an operator-facing summary placeholder."
    $summary += "- Replace or expand this with a handoff summary after review."
    $summary += "- Recommended follow-up: ask Copilot to summarize the imported history into a project handoff."
    $summary += ""

    Set-Content -Path $summaryPath -Value $summary -Encoding UTF8
    return $summaryPath
}

function Split-TextFileIntoParts {
    param(
        [Parameter(Mandatory = $true)]
        [string]$SourceFile,

        [Parameter(Mandatory = $true)]
        [string]$OutputRoot,

        [int]$LinesPerPart = 2000
    )

    if (-not (Test-Path $SourceFile)) {
        return $null
    }

    $baseName = [System.IO.Path]::GetFileNameWithoutExtension($SourceFile)
    $partsFolder = Join-Path $OutputRoot ($baseName + "_parts")

    New-Item -ItemType Directory -Path $partsFolder -Force | Out-Null

    Get-ChildItem -Path $partsFolder -ErrorAction SilentlyContinue |
        Where-Object { -not $_.PSIsContainer -and $_.Name -like "${baseName}_part_*.txt" } |
        Remove-Item -Force -ErrorAction SilentlyContinue

    $chunk = New-Object System.Collections.Generic.List[string]
    $partNumber = 1

    Get-Content -Path $SourceFile | ForEach-Object {
        $chunk.Add($_)

        if ($chunk.Count -ge $LinesPerPart) {
            $partPath = Join-Path $partsFolder ("{0}_part_{1:D3}.txt" -f $baseName, $partNumber)
            $chunk | Set-Content -Path $partPath -Encoding UTF8
            $chunk = New-Object System.Collections.Generic.List[string]
            $partNumber++
        }
    }

    if ($chunk.Count -gt 0) {
        $partPath = Join-Path $partsFolder ("{0}_part_{1:D3}.txt" -f $baseName, $partNumber)
        $chunk | Set-Content -Path $partPath -Encoding UTF8
    }

    return $partsFolder
}

function Get-MonthRecoveryFolderName {
    param([datetime]$Date)

    $monthFolder = $Date.ToString("MMM-yyyy").ToLower()

    return switch -Regex ($monthFolder) {
        '^jan-' { $monthFolder -replace '^jan', 'january' }
        '^feb-' { $monthFolder -replace '^feb', 'february' }
        '^mar-' { $monthFolder -replace '^mar', 'march' }
        '^apr-' { $monthFolder -replace '^apr', 'april' }
        '^may-' { $monthFolder -replace '^may', 'may' }
        '^jun-' { $monthFolder -replace '^jun', 'june' }
        '^jul-' { $monthFolder -replace '^jul', 'july' }
        '^aug-' { $monthFolder -replace '^aug', 'august' }
        '^sep-' { $monthFolder -replace '^sep', 'september' }
        '^oct-' { $monthFolder -replace '^oct', 'october' }
        '^nov-' { $monthFolder -replace '^nov', 'november' }
        '^dec-' { $monthFolder -replace '^dec', 'december' }
        default { $monthFolder }
    }
}

function Assert-GitRepo {
    param([string]$Path)

    if (-not (Test-Path (Join-Path $Path ".git"))) {
        throw "RepoPath is not a Git repository: $Path"
    }
}

    }

if (-not $Silent) {
    Start-Process notepad.exe $masterHistory
}

    try {
        git checkout $BranchName
        if ($LASTEXITCODE -ne 0) { throw "Failed to checkout branch $BranchName" }

        git pull origin $BranchName
        if ($LASTEXITCODE -ne 0) { throw "Failed to pull branch $BranchName" }

            exit 0
        }

        if ($LASTEXITCODE -ne 0) { throw "Git commit failed." }

        git push origin $BranchName
        if ($LASTEXITCODE -ne 0) { throw "Git push failed." }

    }
    finally {
        Pop-Location
    }
}
else {
    Write-Info "SkipGit specified; files prepared but not committed/pushed."
}
