        Dim s As String = If(value, "").Trim()

        If Integer.TryParse(s, yearNum) Then
            If s.Length <= 2 Then
                If yearNum <= 79 Then
                    Return (2000 + yearNum).ToString()
                Else
                    Return (1900 + yearNum).ToString()
                End If
            End If
