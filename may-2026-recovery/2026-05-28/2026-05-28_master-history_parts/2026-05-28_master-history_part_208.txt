    Private Sub HandleLogBookYearPromptKey(rawChar As String)
        ' rawChar is the original KeyPress char (preserve digits)
        If rawChar Is Nothing Then Return

        Dim c As Char = rawChar(0)

        ' Enter -> accept
        If c = ChrW(13) OrElse c = ChrW(10) Then
            Dim year As Integer = _logBookDefaultYear

            Dim typed As String = _logBookYearInput.ToString().Trim()
            If typed.Length > 0 Then
                Dim parsed As Integer
                If Integer.TryParse(typed, parsed) Then
                    year = parsed
                Else
                    ' Invalid input -> ignore (DOS would usually just keep waiting)
                    Return
                End If
            End If

            StopPrompt()
            OpenLogBookForYear(year)
            Return
        End If

        ' Backspace editing
        If c = ChrW(8) Then
            If _logBookYearInput.Length > 0 Then
                _logBookYearInput.Length -= 1
                UpdateLogBookYearPromptText()
            End If
            Return
        End If

        ' Allow digits only (up to 4)
        If Char.IsDigit(c) Then
            If _logBookYearInput.Length < 4 Then
                _logBookYearInput.Append(c)
                UpdateLogBookYearPromptText()
            End If
            Return
        End If

        ' Optional: allow Q to cancel like other prompts
        If c = "q"c OrElse c = "Q"c Then
            StopPrompt()
            Return
        End If

        ' Ignore everything else
    End Sub
