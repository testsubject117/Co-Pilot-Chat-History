## Direct answer

- not the displayed invoice number

2. decode the stored invoice-index field

''' <summary>
''' Reads random-access binary records from INVOICE.CHK.
'''
''' Legacy GW-BASIC behavior from ENTIRE.ASC:
'''   FILNUM = INT(InvoiceNumber - 75000!)
'''   GET #3, FILNUM
'''   IF CVS(INUM$) <> FILNUM THEN "DOESN'T EXIST"
'''
''' In live data, the first field does not decode cleanly with the current
''' .NET MBF conversion routine, but amount/company/flag data does appear
''' structurally correct at the expected slot. Therefore this reader uses
''' direct record positioning and treats the first field as advisory only.
''' </summary>
Public NotInheritable Class InvoiceChkReader
    Private Sub New()
    End Sub

    Private Const RecordSize As Integer = 26
    Private Const InvoiceBase As Long = 75000L

    Public Shared Function DecodeMbfSingle(buf() As Byte, offset As Integer) As Single
        If buf Is Nothing OrElse offset < 0 OrElse offset + 4 > buf.Length Then Return 0.0F

        Dim b0 As Byte = buf(offset)
        Dim b1 As Byte = buf(offset + 1)
        Dim b2 As Byte = buf(offset + 2)
        Dim b3 As Byte = buf(offset + 3)

        If b3 = 0 Then Return 0.0F

        Dim signBit As UInteger = (CUInt(b2) And &H80UI) << 24
        Dim ieeeExp As UInteger = CUInt(b3) - 1UI
        Dim mantissa As UInteger = ((CUInt(b2) And &H7FUI) << 16) Or
                                   (CUInt(b1) << 8) Or
                                   CUInt(b0)

        Dim bits As UInteger = signBit Or (ieeeExp << 23) Or mantissa

        Dim ieee4(3) As Byte
        ieee4(0) = CByte(bits And &HFFUI)
        ieee4(1) = CByte((bits >> 8) And &HFFUI)
        ieee4(2) = CByte((bits >> 16) And &HFFUI)
        ieee4(3) = CByte((bits >> 24) And &HFFUI)

        Dim result As Single = BitConverter.ToSingle(ieee4, 0)
        If Single.IsNaN(result) OrElse Single.IsInfinity(result) Then Return 0.0F
        Return result
    End Function

    Public Shared Function ReadRecord(path As String, invoiceNumber As Long) As InvoiceChkRecord
        Dim rec As New InvoiceChkRecord() With {.InvoiceNumber = invoiceNumber}

        If invoiceNumber <= 0 Then Return rec
        If Not File.Exists(path) Then Return rec

        Dim recordNumber As Long = invoiceNumber - InvoiceBase
        If recordNumber < 1 Then Return rec

        Try
            Using fs As New FileStream(path, FileMode.Open, FileAccess.Read, FileShare.ReadWrite)
                Dim recordOffset As Long = (recordNumber - 1) * RecordSize
                If recordOffset < 0 OrElse recordOffset + RecordSize > fs.Length Then
                    Return rec
                End If

                Dim buf(RecordSize - 1) As Byte
                fs.Seek(recordOffset, SeekOrigin.Begin)

                Dim bytesRead As Integer = fs.Read(buf, 0, RecordSize)
                If bytesRead < RecordSize Then
                    Return rec
                End If

                ' Advisory only; do not use for hard validation.
                Dim storedIndexRaw As Single = DecodeMbfSingle(buf, 0)

                ' Amount field (offset 9) is what the DOS code reads with CVS(AMT$)
                rec.Amount = CDec(DecodeMbfSingle(buf, 9))

                ' Company code (offsets 17-24)
                rec.CompanyCode = Encoding.ASCII.GetString(buf, 17, 8).TrimEnd(ChrW(0), " "c)

                ' Status flag (offset 25)
                Dim flagByte As Byte = buf(25)
                If flagByte <> 0 AndAlso flagByte <> 32 Then
                    rec.Flag = ChrW(flagByte).ToString().ToUpperInvariant()
                Else
                    rec.Flag = ""
                End If

                ' Record existence heuristic:
                ' If company or flag has meaningful content, or amount is non-zero,
                ' consider the record present.
                If rec.CompanyCode <> "" OrElse rec.Flag <> "" OrElse rec.Amount <> 0D OrElse storedIndexRaw <> 0.0F Then
                    rec.IsFound = True
                End If
            End Using
        Catch
            ' Treat any I/O or format issue as not found.
        End Try

        Return rec
    End Function
End Class

''' <summary>Decoded record from INVOICE.CHK.</summary>
Public Class InvoiceChkRecord

Imports System.IO

''' <summary>
''' Reads random-access binary records from INVOICE.CHK.
'''
''' Legacy GW-BASIC behavior from ENTIRE.ASC:
'''   FILNUM = INT(InvoiceNumber - 75000!)
'''   GET #3, FILNUM
'''   IF CVS(INUM$) <> FILNUM THEN "DOESN'T EXIST"
'''
''' In live data, the first field does not decode cleanly with the current
''' .NET MBF conversion routine, but amount/company/flag data does appear
''' structurally correct at the expected slot. Therefore this reader uses
''' direct record positioning and treats the first field as advisory only.
''' </summary>
Public NotInheritable Class InvoiceChkReader
    Private Sub New()
    End Sub

    Private Const RecordSize As Integer = 26
    Private Const InvoiceBase As Long = 75000L

    Public Shared Function DecodeMbfSingle(buf() As Byte, offset As Integer) As Single
        If buf Is Nothing OrElse offset < 0 OrElse offset + 4 > buf.Length Then Return 0.0F

        Dim b0 As Byte = buf(offset)
        Dim b1 As Byte = buf(offset + 1)
        Dim b2 As Byte = buf(offset + 2)
        Dim b3 As Byte = buf(offset + 3)

        If b3 = 0 Then Return 0.0F

        Dim signBit As UInteger = (CUInt(b2) And &H80UI) << 24
        Dim ieeeExp As UInteger = CUInt(b3) - 1UI
        Dim mantissa As UInteger = ((CUInt(b2) And &H7FUI) << 16) Or
                                   (CUInt(b1) << 8) Or
                                   CUInt(b0)

        Dim bits As UInteger = signBit Or (ieeeExp << 23) Or mantissa

        Dim ieee4(3) As Byte
        ieee4(0) = CByte(bits And &HFFUI)
        ieee4(1) = CByte((bits >> 8) And &HFFUI)
        ieee4(2) = CByte((bits >> 16) And &HFFUI)
        ieee4(3) = CByte((bits >> 24) And &HFFUI)

        Dim result As Single = BitConverter.ToSingle(ieee4, 0)
        If Single.IsNaN(result) OrElse Single.IsInfinity(result) Then Return 0.0F
        Return result
    End Function

    Public Shared Function ReadRecord(path As String, invoiceNumber As Long) As InvoiceChkRecord
        Dim rec As New InvoiceChkRecord() With {.InvoiceNumber = invoiceNumber}

        If invoiceNumber <= 0 Then Return rec
        If Not File.Exists(path) Then Return rec

        Dim recordNumber As Long = invoiceNumber - InvoiceBase
        If recordNumber < 1 Then Return rec

        Try
            Using fs As New FileStream(path, FileMode.Open, FileAccess.Read, FileShare.ReadWrite)
                Dim recordOffset As Long = (recordNumber - 1) * RecordSize
                If recordOffset < 0 OrElse recordOffset + RecordSize > fs.Length Then
                    Return rec
                End If

                Dim buf(RecordSize - 1) As Byte
                fs.Seek(recordOffset, SeekOrigin.Begin)

                Dim bytesRead As Integer = fs.Read(buf, 0, RecordSize)
                If bytesRead < RecordSize Then
                    Return rec
                End If

                ' Advisory only; do not use for hard validation.
                Dim storedIndexRaw As Single = DecodeMbfSingle(buf, 0)

                ' Amount field (offset 9) is what the DOS code reads with CVS(AMT$)
                rec.Amount = CDec(DecodeMbfSingle(buf, 9))

                ' Company code (offsets 17-24)
                rec.CompanyCode = Encoding.ASCII.GetString(buf, 17, 8).TrimEnd(ChrW(0), " "c)

                ' Status flag (offset 25)
                Dim flagByte As Byte = buf(25)
                If flagByte <> 0 AndAlso flagByte <> 32 Then
                    rec.Flag = ChrW(flagByte).ToString().ToUpperInvariant()
                Else
                    rec.Flag = ""
                End If

                ' Record existence heuristic:
                ' If company or flag has meaningful content, or amount is non-zero,
                ' consider the record present.
                If rec.CompanyCode <> "" OrElse rec.Flag <> "" OrElse rec.Amount <> 0D OrElse storedIndexRaw <> 0.0F Then
                    rec.IsFound = True
                End If
            End Using
        Catch
            ' Treat any I/O or format issue as not found.
        End Try

        Return rec
    End Function
End Class
