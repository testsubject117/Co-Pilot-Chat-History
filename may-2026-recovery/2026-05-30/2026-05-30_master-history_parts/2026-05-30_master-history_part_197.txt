```vb name=RolodexZipKeyService.vb
Public Function Search(input As String, Optional citySearchText As String = Nothing) As List(Of ZipKeyEntry)
    Dim results As New List(Of ZipKeyEntry)

    If String.IsNullOrWhiteSpace(input) Then
        Return results
    End If

    Dim trimmed = input.Trim()

    If trimmed = "?" Then
        If String.IsNullOrWhiteSpace(citySearchText) Then
            Return results
        End If

        Dim cityTerm = citySearchText.Trim().ToUpperInvariant()

        results = _entries.
            Where(Function(x) Not String.IsNullOrWhiteSpace(x.City) AndAlso
                              x.City.ToUpperInvariant().Contains(cityTerm)).
            OrderBy(Function(x) x.State).
            ThenBy(Function(x) x.City).
            ThenBy(Function(x) x.ZipCode).
            ToList()

        Return results
    End If

    If trimmed.Length = 2 AndAlso trimmed.All(Function(c) Char.IsLetter(c)) Then
        Dim stateCode = trimmed.ToUpperInvariant()

        results = _entries.
            Where(Function(x) String.Equals(x.State, stateCode, StringComparison.OrdinalIgnoreCase)).
            OrderBy(Function(x) x.City).
            ThenBy(Function(x) x.ZipCode).
            ToList()

        Return results
    End If

    Dim digitsOnly = New String(trimmed.Where(Function(c) Char.IsDigit(c)).ToArray())

    If digitsOnly.Length > 0 Then
        results = _entries.
            Where(Function(x) Not String.IsNullOrWhiteSpace(x.ZipCode) AndAlso
                              x.ZipCode.StartsWith(digitsOnly, StringComparison.OrdinalIgnoreCase)).
            OrderBy(Function(x) x.ZipCode).
            ThenBy(Function(x) x.City).
            ThenBy(Function(x) x.State).
            ToList()

        Return results
    End If

    Dim citySearch = trimmed.ToUpperInvariant()

    results = _entries.
        Where(Function(x) Not String.IsNullOrWhiteSpace(x.City) AndAlso
                          x.City.ToUpperInvariant().Contains(citySearch)).
        OrderBy(Function(x) x.State).
        ThenBy(Function(x) x.City).
        ThenBy(Function(x) x.ZipCode).
```
```vb name=RolodexAreaCodeService.vb
        ToList()

    Return results
End Function
```

Imports System.Collections.Generic
Imports System.IO
Imports System.Linq
Imports System.Text

Public Class RolodexAreaCodeService

    Private Shared ReadOnly CsvPath As String = "\\invoice\mainmenu\data\phone\areacodes.csv"
    Private ReadOnly _entries As List(Of AreaCodeEntry)

    Public Sub New()
        _entries = LoadEntries()
    End Sub

    Public Function Search(input As String) As List(Of AreaCodeEntry)
        Dim results As New List(Of AreaCodeEntry)

        If String.IsNullOrWhiteSpace(input) Then
            Return results
        End If

        Dim trimmed = input.Trim()

        If trimmed.Length = 2 AndAlso trimmed.All(Function(c) Char.IsLetter(c)) Then
            Dim stateCode = trimmed.ToUpperInvariant()

            results = _entries.
                Where(Function(x) String.Equals(x.State, stateCode, StringComparison.OrdinalIgnoreCase)).
                OrderBy(Function(x) x.AreaCode).
                ThenBy(Function(x) x.Location).
                ToList()

            Return results
        End If

        Dim digitsOnly = New String(trimmed.Where(Function(c) Char.IsDigit(c)).ToArray())

        If digitsOnly.Length > 0 Then
            results = _entries.
                Where(Function(x) Not String.IsNullOrWhiteSpace(x.AreaCode) AndAlso
                                  x.AreaCode.StartsWith(digitsOnly, StringComparison.OrdinalIgnoreCase)).
                OrderBy(Function(x) x.AreaCode).
                ThenBy(Function(x) x.State).
                ThenBy(Function(x) x.Location).
                ToList()

            Return results
        End If

        Dim locationSearch = trimmed.ToUpperInvariant()

        results = _entries.
            Where(Function(x) Not String.IsNullOrWhiteSpace(x.Location) AndAlso
                              x.Location.ToUpperInvariant().Contains(locationSearch)).
            OrderBy(Function(x) x.State).
            ThenBy(Function(x) x.Location).
            ThenBy(Function(x) x.AreaCode).
            ToList()

        Return results
    End Function

    Public Function GetAllPages(Optional linesPerPage As Integer = 35) As List(Of String)
        Dim pages As New List(Of String)

        Dim ordered = _entries.
            OrderBy(Function(x) x.AreaCode).
            ThenBy(Function(x) x.State).
            ThenBy(Function(x) x.Location).
            ToList()

        If ordered.Count = 0 Then
            Return pages
        End If

        If linesPerPage <= 0 Then
            linesPerPage = 35
        End If

        For i As Integer = 0 To ordered.Count - 1 Step linesPerPage
            Dim pageEntries = ordered.Skip(i).Take(linesPerPage).ToList()
            Dim sb As New StringBuilder()

            sb.AppendLine("Area Code Directory")
            sb.AppendLine(New String("-"c, 46))
            sb.AppendLine("Code   State  Location")
            sb.AppendLine(New String("-"c, 46))

            For Each item In pageEntries
                sb.AppendLine(item.AreaCode.PadRight(6) &
                              item.State.PadRight(7) &
                              item.Location)
            Next

            pages.Add(sb.ToString())
        Next

        Return pages
    End Function

    Public Function DataFileExists() As Boolean
        Return File.Exists(CsvPath)
    End Function

    Public Function GetDataFilePath() As String
        Return CsvPath
    End Function
