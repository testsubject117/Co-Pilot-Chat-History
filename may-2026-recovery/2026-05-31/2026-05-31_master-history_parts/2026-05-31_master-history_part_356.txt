        pageSb.AppendLine()
        pageEntryCount += 1
    End Sub

    Private Sub FlushPageIfNeeded(pages As List(Of String),
                                 pageSb As System.Text.StringBuilder,
                                 ByRef pageEntryCount As Integer)
        If pageSb.Length > 0 Then
            pages.Add(pageSb.ToString().TrimEnd())
            pageSb.Clear()
            pageEntryCount = 0
        End If
    End Sub

End Class
