Option Strict On
Option Explicit On

Imports System.IO
Imports System.Text

''' <summary>
''' Reads random-access binary records from INVOICE.CHK.
''' Each record is 26 bytes.
''' Legacy GW-BASIC layout:
'''   INUM$ : 9 bytes  (MBF single stored in first 4 bytes, remaining bytes padding)
'''   AMT$  : 8 bytes  (MBF single stored in first 4 bytes, remaining bytes padding)
'''   CO$   : 8 bytes  (ASCII company code)
'''   FLAG$ : 1 byte   (status character: J/P/C/V/E)
''' </summary>
Public NotInheritable Class InvoiceChkReader
    Private Sub New()
    End Sub

    Private Const RecordSize As Integer = 26
    Private Const InvoiceBase As Long = 75000L

    Public Shared Function DecodeMbfSingle(buf() As Byte, offset As Integer) As Single
        If buf Is Nothing OrElse offset < 0 OrElse offset + 4 > buf.Length Then Return 0.0F

        Dim b0 As Byte = buf(offset)
        Dim b1 As Byte = buf(offset + 1)
        Dim b2 As Byte = buf(offset + 2)
        Dim b3 As Byte = buf(offset + 3)

        If b3 = 0 Then Return 0.0F

        Dim signBit As UInteger = (CUInt(b2) And &H80UI) << 24
        Dim ieeeExp As UInteger = CUInt(b3) - 1UI
        Dim mantissa As UInteger = ((CUInt(b2) And &H7FUI) << 16) Or
                                   (CUInt(b1) << 8) Or
                                   CUInt(b0)

        Dim bits As UInteger = signBit Or (ieeeExp << 23) Or mantissa

        Dim ieee4(3) As Byte
        ieee4(0) = CByte(bits And &HFFUI)
        ieee4(1) = CByte((bits >> 8) And &HFFUI)
        ieee4(2) = CByte((bits >> 16) And &HFFUI)
        ieee4(3) = CByte((bits >> 24) And &HFFUI)

        Dim result As Single = BitConverter.ToSingle(ieee4, 0)
        If Single.IsNaN(result) OrElse Single.IsInfinity(result) Then Return 0.0F
        Return result
    End Function

    Public Shared Function ReadRecord(path As String, invoiceNumber As Long) As InvoiceChkRecord
        Dim emptyResult As New InvoiceChkRecord() With {.InvoiceNumber = invoiceNumber}

        If invoiceNumber <= 0 Then Return emptyResult
        If Not File.Exists(path) Then Return emptyResult

        Dim fileSize As Long
        Try
            fileSize = New FileInfo(path).Length
        Catch
            Return emptyResult
        End Try

        If fileSize < RecordSize Then Return emptyResult

        ' GW-BASIC uses record number = invoice - 75000
        Dim directRecordNumber As Long = invoiceNumber - InvoiceBase
        If directRecordNumber < 1 Then Return emptyResult

        Try
            Using fs As New FileStream(path, FileMode.Open, FileAccess.Read, FileShare.ReadWrite)
                ' First try the exact direct slot.
                Dim direct = ReadRecordAtRecordNumber(fs, invoiceNumber, directRecordNumber)
                If direct.IsFound Then
                    Return direct
                End If

                ' Fallback: search a very small nearby window to tolerate historical drift.
                ' This keeps lookups fast while handling possible off-by-one / nearby-slot anomalies.
                Dim delta As Integer
                For delta = -3 To 3
                    If delta = 0 Then Continue For

                    Dim candidateRecordNumber As Long = directRecordNumber + delta
                    If candidateRecordNumber < 1 Then Continue For

                    Dim candidate = ReadRecordAtRecordNumber(fs, invoiceNumber, candidateRecordNumber)
                    If candidate.IsFound Then
                        Return candidate
                    End If
                Next
            End Using
        Catch
            ' Treat any I/O/format failure as not found.
        End Try

        Return emptyResult
    End Function

    Private Shared Function ReadRecordAtRecordNumber(fs As FileStream, requestedInvoiceNumber As Long, recordNumber As Long) As InvoiceChkRecord
        Dim rec As New InvoiceChkRecord() With {.InvoiceNumber = requestedInvoiceNumber}

        If fs Is Nothing Then Return rec
        If recordNumber < 1 Then Return rec

        Dim recordOffset As Long = (recordNumber - 1) * RecordSize
        If recordOffset < 0 OrElse recordOffset + RecordSize > fs.Length Then Return rec

        Dim buf(RecordSize - 1) As Byte

        fs.Seek(recordOffset, SeekOrigin.Begin)
        Dim bytesRead As Integer = fs.Read(buf, 0, RecordSize)
        If bytesRead < RecordSize Then Return rec

        Dim storedInvoiceRaw As Single = DecodeMbfSingle(buf, 0)
        Dim storedAmountRaw As Single = DecodeMbfSingle(buf, 9)

        Dim storedInvoiceNumber As Long = CLng(Math.Round(CDbl(storedInvoiceRaw), MidpointRounding.AwayFromZero))

        ' Validate against the requested invoice number.
        ' If it doesn't match, this isn't the record we want.
        If storedInvoiceNumber <> requestedInvoiceNumber Then
            Return rec
        End If

        rec.Amount = CDec(storedAmountRaw)

        rec.CompanyCode = Encoding.ASCII.GetString(buf, 17, 8).TrimEnd(ChrW(0), " "c)

        Dim flagByte As Byte = buf(25)
        If flagByte <> 0 AndAlso flagByte <> 32 Then
            rec.Flag = ChrW(flagByte).ToString().ToUpperInvariant()
        Else
            rec.Flag = ""
        End If

        rec.IsFound = True
        Return rec
    End Function
End Class

Imports System.IO
Imports System.Text

''' <summary>
''' Reads random-access binary records from INVOICE.CHK.
''' Actual field decoding is valid at 26 bytes per record:
'''   INUM$ : 9 bytes  (MBF single in first 4 bytes)
'''   AMT$  : 8 bytes  (MBF single in first 4 bytes)
'''   CO$   : 8 bytes  (ASCII company code)
'''   FLAG$ : 1 byte   (status character)
'''
''' Although legacy notes suggest direct slot math from invoice number,
''' live data may not follow that mapping consistently. To remain robust,
''' this reader scans records until it finds the requested invoice number.
''' </summary>
Public NotInheritable Class InvoiceChkReader
    Private Sub New()
    End Sub

    Private Const RecordSize As Integer = 26

    Public Shared Function DecodeMbfSingle(buf() As Byte, offset As Integer) As Single
        If buf Is Nothing OrElse offset < 0 OrElse offset + 4 > buf.Length Then Return 0.0F

        Dim b0 As Byte = buf(offset)
        Dim b1 As Byte = buf(offset + 1)
        Dim b2 As Byte = buf(offset + 2)
        Dim b3 As Byte = buf(offset + 3)

        If b3 = 0 Then Return 0.0F

        Dim signBit As UInteger = (CUInt(b2) And &H80UI) << 24
        Dim ieeeExp As UInteger = CUInt(b3) - 1UI
        Dim mantissa As UInteger = ((CUInt(b2) And &H7FUI) << 16) Or
                                   (CUInt(b1) << 8) Or
                                   CUInt(b0)

        Dim bits As UInteger = signBit Or (ieeeExp << 23) Or mantissa

        Dim ieee4(3) As Byte
        ieee4(0) = CByte(bits And &HFFUI)
        ieee4(1) = CByte((bits >> 8) And &HFFUI)
        ieee4(2) = CByte((bits >> 16) And &HFFUI)
        ieee4(3) = CByte((bits >> 24) And &HFFUI)

        Dim result As Single = BitConverter.ToSingle(ieee4, 0)
        If Single.IsNaN(result) OrElse Single.IsInfinity(result) Then Return 0.0F
        Return result
    End Function

    Public Shared Function ReadRecord(path As String, invoiceNumber As Long) As InvoiceChkRecord
        Dim result As New InvoiceChkRecord() With {.InvoiceNumber = invoiceNumber}

        If invoiceNumber <= 0 Then Return result
        If Not File.Exists(path) Then Return result

        Try
            Using fs As New FileStream(path, FileMode.Open, FileAccess.Read, FileShare.ReadWrite)
                Dim fileLength As Long = fs.Length
                If fileLength < RecordSize Then
                    Return result
                End If

                Dim recordCount As Long = fileLength \ RecordSize
                Dim recordIndex As Long

                For recordIndex = 0 To recordCount - 1
                    Dim rec = ReadRecordAtIndex(fs, recordIndex)
                    If rec.IsFound AndAlso rec.InvoiceNumber = invoiceNumber Then
                        Return rec
                    End If
                Next
            End Using
        Catch
            ' Treat any error as not found.
        End Try

        Return result
    End Function

    Private Shared Function ReadRecordAtIndex(fs As FileStream, recordIndex As Long) As InvoiceChkRecord
        Dim rec As New InvoiceChkRecord()

        If fs Is Nothing Then Return rec
        If recordIndex < 0 Then Return rec

        Dim recordOffset As Long = recordIndex * RecordSize
        If recordOffset < 0 OrElse recordOffset + RecordSize > fs.Length Then Return rec

        Dim buf(RecordSize - 1) As Byte
        fs.Seek(recordOffset, SeekOrigin.Begin)

        Dim bytesRead As Integer = fs.Read(buf, 0, RecordSize)
        If bytesRead < RecordSize Then Return rec

        Dim storedInvoiceRaw As Single = DecodeMbfSingle(buf, 0)
        Dim storedAmountRaw As Single = DecodeMbfSingle(buf, 9)

        Dim storedInvoiceNumber As Long = CLng(Math.Round(CDbl(storedInvoiceRaw), MidpointRounding.AwayFromZero))

        ' Skip clearly empty/invalid records
        If storedInvoiceNumber <= 0 Then
            Return rec
        End If

        rec.InvoiceNumber = storedInvoiceNumber
        rec.Amount = CDec(storedAmountRaw)
        rec.CompanyCode = Encoding.ASCII.GetString(buf, 17, 8).TrimEnd(ChrW(0), " "c)

        Dim flagByte As Byte = buf(25)
        If flagByte <> 0 AndAlso flagByte <> 32 Then
            rec.Flag = ChrW(flagByte).ToString().ToUpperInvariant()
        Else
            rec.Flag = ""
        End If

        rec.IsFound = True
        Return rec
    End Function
End Class

''' <summary>Decoded record from INVOICE.CHK.</summary>
Public Class InvoiceChkRecord
    Public Property InvoiceNumber As Long
    Public Property Amount As Decimal
    Public Property CompanyCode As String = ""
    Public Property Flag As String = ""
    Public Property IsFound As Boolean = False

    Public ReadOnly Property FlagDescription As String
        Get
            Select Case Flag
                Case "J" : Return "Open"
                Case "P" : Return "Paid"
                Case "C" : Return "Corrected Cert"
                Case "V" : Return "Void"
                Case "E" : Return "Erased"
                Case Else : Return If(Flag, "")
            End Select
        End Get
    End Property
End Class

''' <summary>
''' Flat row bound to the invoice-details DataGridView inside FormLedgerView.
''' </summary>
Public Class InvoiceDetailRow
    Public Property InvoiceNumber As Long
    Public Property Amount As String = ""
    Public Property CompanyCode As String = ""
    Public Property Status As String = ""
End Class

Option Strict On

Imports System.Globalization
Imports System.IO
Imports System.Text

Public NotInheritable Class CheckInvReader
    Private Sub New()
    End Sub

    Public Shared Function ReadAll(path As String) As List(Of CheckInvBlock)
        Dim results As New List(Of CheckInvBlock)()
        If Not File.Exists(path) Then Return results

        Dim rawLines As String() = File.ReadAllLines(path)
        Dim i As Integer = 0

        While i < rawLines.Length
            Dim line As String = CleanLine(rawLines(i))

            If line.Length = 0 Then
                i += 1
                Continue While
            End If

            ' Header lines are GW-BASIC WRITE# output and start with a quote.
            If Not line.StartsWith("""", StringComparison.Ordinal) Then
                i += 1
                Continue While
            End If

            Dim header As CheckInvBlock = ParseHeader(line)
            i += 1

            ' Collect invoice numbers until we reach the invoice count or the next header.
            Dim invoices As New List(Of String)()

            ' Defensive: bad/blank counts happen
            Dim target As Integer = header.InvoiceCount
            If target < 0 Then target = 0
            If target > 5000 Then target = 5000 ' safety cap

            While i < rawLines.Length AndAlso invoices.Count < target
                Dim invLine As String = CleanLine(rawLines(i))

                If invLine.Length = 0 Then
                    i += 1
                    Continue While
                End If

                ' If we hit the next header early, stop this block.
                If invLine.StartsWith("""", StringComparison.Ordinal) Then
                    Exit While
                End If

                ' A single line may contain one invoice or many invoices.
                ' Try to parse as WRITE#-style CSV first. If it yields multiple parts, use them.
                Dim parts As List(Of String) = SplitBasicWriteLine(invLine)

                If parts.Count > 1 Then
                    For Each p In parts
                        Dim v = NormalizeInvoiceToken(p)
                        If v.Length > 0 Then invoices.Add(v)
                        If invoices.Count >= target Then Exit For
                    Next
                Else
                    ' Not a CSV-ish line; treat as a single invoice token
                    Dim v = NormalizeInvoiceToken(invLine)
                    If v.Length > 0 Then invoices.Add(v)
                End If

                i += 1
            End While

            header.Invoices = invoices
            results.Add(header)
        End While

        Return results
    End Function
    Public Shared Function DiagnoseFormat(path As String) As String
        If Not File.Exists(path) Then Return "File not found: " & path

        Dim rawLines = File.ReadAllLines(path)
        Dim headers As Integer = 0
        Dim invLines As Integer = 0
        Dim invLinesWithCommas As Integer = 0
        Dim invLinesThatLookQuoted As Integer = 0

        Dim seen As Integer = 0
        For Each raw In rawLines
            Dim line = If(raw, "").Replace(ChrW(26), "").Trim()
            If line = "" Then Continue For

            seen += 1
            If line.StartsWith("""", StringComparison.Ordinal) Then
                headers += 1
            Else
                invLines += 1
                If line.Contains(",", StringComparison.Ordinal) Then invLinesWithCommas += 1
                If line.StartsWith("""", StringComparison.Ordinal) OrElse line.Contains("""", StringComparison.Ordinal) Then invLinesThatLookQuoted += 1
            End If

            If seen >= 200 Then Exit For
        Next

        Return $"Lines scanned: {seen}" & Environment.NewLine &
           $"Header lines (start with quote): {headers}" & Environment.NewLine &
           $"Non-header lines: {invLines}" & Environment.NewLine &
           $"Non-header lines containing commas: {invLinesWithCommas}" & Environment.NewLine &
           $"Non-header lines containing quotes: {invLinesThatLookQuoted}"
    End Function
    Private Shared Function ParseHeader(line As String) As CheckInvBlock
        ' Example:
        ' "NATAIRCR","8508","4","01-03-1995",196.75,4
        Dim parts As List(Of String) = SplitBasicWriteLine(line)

        Dim b As New CheckInvBlock() With {
            .CustomerCode = GetPart(parts, 0),
            .CheckNumber = GetPart(parts, 1),
            .SalesmanCode = GetPart(parts, 2),
            .DateText = GetPart(parts, 3),
            .Amount = ParseDecimal(GetPart(parts, 4)),
            .InvoiceCount = ParseInt(GetPart(parts, 5)),
            .Invoices = New List(Of String)()
        }

        Return b
    End Function

    Private Shared Function CleanLine(s As String) As String
        If s Is Nothing Then Return ""
        Return s.Replace(ChrW(26), "").Trim()
    End Function

    Private Shared Function NormalizeInvoiceToken(s As String) As String
        If s Is Nothing Then Return ""
        Dim t = s.Replace(ChrW(26), "").Trim()

        ' Some legacy files may quote invoice numbers even on "invoice lines"
        ' SplitBasicWriteLine already unquotes, but for safety:
        If t.Length >= 2 AndAlso t.StartsWith("""", StringComparison.Ordinal) AndAlso t.EndsWith("""", StringComparison.Ordinal) Then
            t = t.Substring(1, t.Length - 2).Trim()
        End If

        ' Optional normalization that often matches legacy: collapse internal spaces
        ' (keep it conservative to avoid changing real invoice formats)
        t = t.Replace(vbTab, " ")
        While t.Contains("  ", StringComparison.Ordinal)
            t = t.Replace("  ", " ")
        End While

        Return t
    End Function

    Private Shared Function GetPart(parts As List(Of String), idx As Integer) As String
        If parts Is Nothing OrElse idx < 0 OrElse idx >= parts.Count Then Return ""
        Return If(parts(idx), "").Trim()
    End Function

    Private Shared Function ParseDecimal(s As String) As Decimal
        If s Is Nothing Then Return 0D
        s = s.Replace(ChrW(26), "").Trim()

        Dim d As Decimal
        If Decimal.TryParse(s, NumberStyles.Any, CultureInfo.InvariantCulture, d) Then Return d
        If Decimal.TryParse(s, NumberStyles.Any, CultureInfo.CurrentCulture, d) Then Return d
        Return 0D
    End Function

    Private Shared Function ParseInt(s As String) As Integer
        If s Is Nothing Then Return 0
        s = s.Replace(ChrW(26), "").Trim()

        Dim n As Integer
        If Integer.TryParse(s, NumberStyles.Integer, CultureInfo.InvariantCulture, n) Then Return n
        If Integer.TryParse(s, NumberStyles.Integer, CultureInfo.CurrentCulture, n) Then Return n
        Return 0
    End Function

    ' - quoted strings allowed
    ' - embedded quotes written as "" inside quoted strings
    ' - numeric fields unquoted
    Private Shared Function SplitBasicWriteLine(line As String) As List(Of String)

        If line Is Nothing Then Return result

        Dim sb As New StringBuilder()
        Dim inQuotes As Boolean = False
        Dim i As Integer = 0

        While i < line.Length
            Dim ch As Char = line(i)

            If inQuotes Then
                If ch = """"c Then
                    ' Doubled quote => literal quote
                    If i + 1 < line.Length AndAlso line(i + 1) = """"c Then
                        sb.Append(""""c)
                        i += 2
                        Continue While
                    Else
                        inQuotes = False
                        i += 1
                        Continue While
                    End If
                Else
                    sb.Append(ch)
                    i += 1
                    Continue While
                End If
            Else
                If ch = """"c Then
                    inQuotes = True
                    i += 1
                    Continue While
                End If

                If ch = ","c Then
                    result.Add(sb.ToString().Trim())
                    sb.Clear()
                    i += 1
                    Continue While
                End If

                sb.Append(ch)
                i += 1
            End If
        End While

        result.Add(sb.ToString().Trim())
        Return result
    End Function
End Class
