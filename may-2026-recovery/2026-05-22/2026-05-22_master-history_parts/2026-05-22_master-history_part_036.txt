    Private Shared Function YearMatches(dateText As String, requestedYear As String) As Boolean
        If String.IsNullOrWhiteSpace(requestedYear) Then Return True
        If String.IsNullOrWhiteSpace(dateText) Then Return False

        If normalizedRequested.Length = 2 Then
            Return dateText.Trim().EndsWith(normalizedRequested, StringComparison.OrdinalIgnoreCase)
        End If

        If normalizedRequested.Length = 4 Then
            Return dateText.Trim().EndsWith(normalizedRequested, StringComparison.OrdinalIgnoreCase) OrElse
                   dateText.Trim().EndsWith(normalizedRequested.Substring(2), StringComparison.OrdinalIgnoreCase)
        End If

        Return dateText.IndexOf(normalizedRequested, StringComparison.OrdinalIgnoreCase) >= 0
    End Function
