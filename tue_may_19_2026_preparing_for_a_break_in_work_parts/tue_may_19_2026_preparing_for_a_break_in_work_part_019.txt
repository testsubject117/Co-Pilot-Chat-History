        _currentCheck = info

        lblLine1.Text = ""

        lblCustomer.Text = "Customer: " & info.CustomerName
        lblDate.Text = "Date: " & info.CheckDate
        lblCheckInfo.Text = "Check #: " & info.CheckNumber & "    Reference: " & info.Reference

        lblStatus2.Text = ""
        lblStatus3.Text = ""

    End Sub

        _state = DeleteCheckState.Deleting

        lblStatus2.Text = ""

        Me.Refresh()
        Application.DoEvents()
    End Sub

    Private Sub txtInput_KeyDown(sender As Object, e As KeyEventArgs) Handles txtInput.KeyDown
        If _state <> DeleteCheckState.EnterCheckNumber Then Return

        If e.KeyCode = Keys.Enter Then
            e.SuppressKeyPress = True

            Dim checkNo As String = txtInput.Text.Trim()

            If checkNo = "" Then
                Me.DialogResult = DialogResult.Cancel
                Me.Close()
                Return
            End If

            Dim found As CheckDeleteInfo = FindCheckByNumber(checkNo)

            If found Is Nothing Then

                txtInput.SelectAll()
                Return
            End If

            ShowCheckDetails(found)
        End If
    End Sub

            End If
