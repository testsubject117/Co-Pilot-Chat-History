            ValidateCompany()

    Private Sub ValidateCompany()
        If _lookup Is Nothing Then Exit Sub

        lstMatches.Items.Clear()
        btnContinue.Enabled = False
        _selectedCompany = Nothing

        Dim enteredCode As String = txtCompanyCode.Text.Trim().ToUpperInvariant()

        If enteredCode = "" Then
            lblStatus.Text = "Company code is required."
            txtCompanyCode.Focus()
            Exit Sub
        End If

        If enteredCode.Length > 8 Then
            enteredCode = enteredCode.Substring(0, 8)
            txtCompanyCode.Text = enteredCode
        End If

        Dim exact As CompanyInfo = _lookup.FindExact(enteredCode)

        If exact IsNot Nothing Then
            _selectedCompany = exact
            lblStatus.Text = exact.CompanyCode & " - " & exact.CompanyName
            btnContinue.Enabled = True
            Exit Sub
        End If

        lblStatus.Text = "Company code not found."

        Dim matches As List(Of CompanyInfo) = _lookup.FindByPrefix(enteredCode)

        If matches.Count = 0 AndAlso enteredCode.Length > 0 Then
            matches = _lookup.FindByFirstLetter(enteredCode.Substring(0, 1))
        End If

        For Each item As CompanyInfo In matches
            lstMatches.Items.Add(item.CompanyCode & " - " & item.CompanyName)
        Next

        If matches.Count > 0 Then
            lblStatus.Text = "Company not found. Select a match or enter another code."
        End If
    End Sub

        ElseIf e.KeyCode = Keys.Enter Then
            If txtCompanyCode.Focused Then
                btnValidate.PerformClick()
                e.Handled = True
            End If
        End If

Public Class FormAddChecks

    Private lblLine1 As Label
    Private lblLine2 As Label
    Private lblPrompt As Label
    Private txtInput As TextBox
    Private btnCancel As Button

    Private Enum AddCheckStep
        CompanyName = 0
        CheckAmount = 1
        LowestInvoice = 2
        HighestInvoice = 3
        Complete = 4
    End Enum

    Private _step As AddCheckStep = AddCheckStep.CompanyName

    Private _checkAmountText As String = ""
    Private _lowestInvoiceText As String = ""
    Private _highestInvoiceText As String = ""

    Public ReadOnly Property CompanyName As String

            Return _companyName
        End Get
    End Property

            Return _checkAmountText

    End Property

    End Property

            Return _highestInvoiceText

    End Property

        lblTitle.Text = "***** Add A Check *****"
        lblTitle.TextAlign = ContentAlignment.MiddleLeft
        lblTitle.Font = New Font("Consolas", 16.0!, FontStyle.Regular)
        lblTitle.ForeColor = Color.White
        lblTitle.BackColor = Color.Black
        lblTitle.SetBounds(24, 20, 700, 32)

        lblLine1 = New Label()
        lblLine1.AutoSize = False
        lblLine1.Text = ""
        lblLine1.TextAlign = ContentAlignment.MiddleLeft
        lblLine1.Font = New Font("Consolas", 12.0!, FontStyle.Regular)
        lblLine1.ForeColor = Color.White
        lblLine1.BackColor = Color.Black
        lblLine1.SetBounds(24, 80, 720, 28)

        lblLine2 = New Label()
        lblLine2.AutoSize = False
        lblLine2.Text = ""

        lblLine2.Font = New Font("Consolas", 12.0!, FontStyle.Regular)

        txtInput.Text = ""
        txtInput.Focus()

        lblLine1.Text = "Company Name: " & _companyName
        lblLine2.Text = "Check Amount ? " & _checkAmountText

            Return

            e.Handled = True
            Return
        End If
    End Sub

    Private Sub ProcessCurrentStep()
        Dim value As String = txtInput.Text.Trim()

        Select Case _step
            Case AddCheckStep.CompanyName
                If String.Equals(value, "Q", StringComparison.OrdinalIgnoreCase) Then
                    btnCancel.PerformClick()
                    Return
                End If

                If value = "" Then
                    txtInput.Focus()
                    Return
                End If

                _companyName = value
                ShowCheckAmountPrompt()

            Case AddCheckStep.CheckAmount
                If value = "" Then
                    txtInput.Focus()
                    Return
                End If

                                    "Add A Check",
                                    MessageBoxButtons.OK,

                    txtInput.SelectAll()
                    Return
                End If

                _checkAmountText = value

                lblLine1.Text = "I need the 1st and last invoice numbers so I can automatically scan for Invoices"
                lblLine2.Text = ""
                ShowLowestInvoicePrompt()

            Case AddCheckStep.LowestInvoice
                If value = "" Then
                    txtInput.Focus()
                    Return
                End If

                _lowestInvoiceText = value

                If value = "-1" Then
                    CompleteFlow()
                    Return
                End If

                ShowHighestInvoicePrompt()

            Case AddCheckStep.HighestInvoice
                If value = "" Then
                    txtInput.Focus()
                    Return
                End If

                _highestInvoiceText = value
                CompleteFlow()
        End Select
    End Sub

Yes — the likely culprit is the `CompanyName` property name.

That warning is not just cosmetic here; in VB/WinForms, naming a property `CompanyName` on a `Form` can collide with inherited members and sometimes trigger weird compiler behavior, especially with newer Roslyn builds and designer-generated metadata. The stack trace also smells like a compiler crash rather than a normal code error.

## Fix
Rename the custom property from:

- `CompanyName`

to something safer, like:

- `EnteredCompanyName`

Also rename the backing field if you want to keep things clean.
