Imports System.Windows.Forms

Public Class FormDeleteCheck

    Private Function CreateLabel(x As Integer, y As Integer, w As Integer, h As Integer) As Label
        Dim lbl As New Label()
        lbl.AutoSize = False
        lbl.Location = New Point(x, y)
        lbl.Size = New Size(w, h)
        lbl.ForeColor = Color.White
        lbl.BackColor = Color.Black
        lbl.Font = New Font("Consolas", 12.0F, FontStyle.Regular)
        Return lbl
    End Function

    Private Sub SetupInitialScreen()
        _state = DeleteCheckState.EnterCheckNumber
        _currentCheck = Nothing

        lblTitle.Text = "***** Delete A Check *****"
        lblLine1.Text = "Check # [ENTER = Quit]?"
        txtInput.Visible = True
        txtInput.Text = ""
        txtInput.Enabled = True
        txtInput.Focus()

        lblCustomer.Text = ""
        lblDate.Text = ""
        lblCheckInfo.Text = ""
        lblInvoices.Text = ""
        lblAmount.Text = ""
        lblPrompt.Text = ""
        lblStatus1.Text = ""
        lblStatus2.Text = ""
        lblStatus3.Text = ""

        lblLine1.Text = ""

        Application.DoEvents()
    End Sub

    Private Sub txtInput_KeyDown(sender As Object, e As KeyEventArgs)
        If _state <> DeleteCheckState.EnterCheckNumber Then Return

        If e.KeyCode = Keys.Enter Then
            e.SuppressKeyPress = True

            Dim checkNo As String = txtInput.Text.Trim()

            If checkNo = "" Then

                Me.Close()
                Return
            End If

            Dim found As CheckDeleteInfo = FindCheckByNumber(checkNo)

                txtInput.SelectAll()
                Return
            End If

            ShowCheckDetails(found)
        End If
    End Sub

    Protected Overrides Function ProcessCmdKey(ByRef msg As Message, keyData As Keys) As Boolean
