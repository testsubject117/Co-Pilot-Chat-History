        End If

            e.Handled = True
            Return
        End If

        Dim value As String = txtInput.Text.Trim()

        Select Case _step
            Case AddCheckStep.CompanyNameStep

                    Return
                End If

                If value = "" Then
                    txtInput.Focus()
                    Return
                End If

                _enteredCompanyName = value

            Case AddCheckStep.CheckAmountStep

                    Return

                                    MessageBoxButtons.OK,
                                    MessageBoxIcon.Information)
                    txtInput.Focus()
                    txtInput.SelectAll()

                End If

                ShowLowestInvoicePrompt()

            Case AddCheckStep.LowestInvoiceStep
                If value = "" Then
                    txtInput.Focus()
                    Return
                End If

                _enteredLowestInvoiceText = value

                If value = "-1" Then
                    CompleteFlow()
                    Return
                End If

                ShowHighestInvoicePrompt()

            Case AddCheckStep.HighestInvoiceStep
                If value = "" Then
                    txtInput.Focus()
                    Return
                End If

                _enteredHighestInvoiceText = value
                CompleteFlow()
        End Select
    End Sub

also, there is a text field on the add a check screen now below C and P and Q, why is that there, what is it for?: 

## Copilot

It’s there because the current WinForms version is using **one shared input textbox for every DOS prompt step**, including the review menu.

## Direct answer
That textbox is currently serving as the place where the user is supposed to type:

- `C`
- `P`
- or `Q`

and then press **Enter**.

So in the current implementation, it is effectively the input field for the review menu choice.

---

## Why it feels odd
Because DOS didn’t visually look like:
- menu options
- then a blank white Windows textbox underneath

DOS was more like:
- text screen
- cursor/prompt inline
- single-character command entry felt native to the screen

In WinForms, the textbox is just the easiest stand-in for that typed input.

---

## What it is for right now
On that review screen, the textbox is for entering:

- `C` → Check is Correct
- `P` → Print list of invoices
- `Q` → Quit to Main Menu

That’s why it appears under the menu text.

---

## Is it ideal?
Not really.

It works, but visually it’s not very DOS-like.

---

## Better options
We have a few ways to improve it:

### Option 1 — keep it
Simplest.
- user types `C`, `P`, or `Q` in the box
- presses Enter

### Option 2 — add an explicit prompt label
Better.
Show something like:

- `Enter Choice [C/P/Q] ?`

above the textbox

That would make the field’s purpose obvious.

### Option 3 — hide the textbox entirely for the review menu and capture keystrokes directly
Most DOS-like.
- pressing `C` immediately advances
- pressing `P` prints
- pressing `Q` quits
- no visible text field needed

This is probably the best DOS-faithful behavior.

---

## My recommendation
I recommend **Option 3** for the review screen:
- no textbox visible there
- just listen for `C`, `P`, `Q` keypresses directly

Then:
- when user picks `C`, show the textbox again for:
  - `Check # (Top Right)?`

That would feel much closer to DOS.

---

## Short answer
That textbox is currently there so the user can type **C, P, or Q**.  
It’s functional, but not ideal. The best next improvement would be to **hide the textbox on that review screen and capture C/P/Q directly from the keyboard**.

    End Sub

    Private Sub ClearLines()
        lblLine1.Text = ""
        lblLine2.Text = ""
        lblLine3.Text = ""
        lblLine4.Text = ""
        lblLine5.Text = ""
    End Sub

    Private Sub ShowInputPrompt(promptText As String)
        lblPrompt.Text = promptText
        txtInput.Visible = True
        txtInput.Text = ""
        txtInput.Focus()
        txtInput.SelectAll()
    End Sub

    Private Sub HideInputPrompt(promptText As String)
        lblPrompt.Text = promptText
        txtInput.Visible = False
        Me.Focus()
    End Sub

    Private Sub ShowCompanyPrompt()
        currentStep = 1
        ClearLines()
        ShowInputPrompt("Company Name [Q = Quit] ?")
    End Sub

    Private Sub ShowAmountPrompt()
        currentStep = 2
        ClearLines()
        lblLine1.Text = "Company Name: " & enteredCompany
        ShowInputPrompt("Check Amount ?")
    End Sub

    Private Sub ShowLowestInvoicePrompt()
        currentStep = 3
        ClearLines()
        lblLine1.Text = "I need the 1st and last invoice numbers so I can automatically scan for invoices."
        ShowInputPrompt("Enter the LOWEST invoice number to be paid [-1 = No Auto Scan] ?")
    End Sub

    Private Sub ShowHighestInvoicePrompt()
        currentStep = 4
        ClearLines()
        lblLine1.Text = "I need the 1st and last invoice numbers so I can automatically scan for invoices."
        lblLine2.Text = "Lowest Invoice Number: " & enteredLowestInvoice
        ShowInputPrompt("Enter the HIGHEST invoice number to be paid ?")
    End Sub

    Private Sub ShowReviewPrompt()
        currentStep = 5
        ClearLines()

        lblLine1.Text = enteredCompany & "  REF:  Check Amount:$  " & enteredAmount & "  { 0 Invoices }"
        lblLine2.Text = "Difference Between Total of Invoices & Check Amount: 0"
        lblLine3.Text = ""
        lblLine4.Text = "(C) Check is Correct, add to the ledger"
        lblLine5.Text = "(P) Print list of invoices"

        HideInputPrompt("(Q) Quit to Main Menu")
    End Sub

    Private Sub ShowCheckNumberPrompt()
        currentStep = 6
        lblLine1.Text = enteredCompany & "  REF:  Check Amount:$  " & enteredAmount & "  { 0 Invoices }"
        lblLine2.Text = "Difference Between Total of Invoices & Check Amount: 0"
        lblLine3.Text = ""
        lblLine4.Text = "(C) Check is Correct, add to the ledger"
        lblLine5.Text = "(P) Print list of invoices"
        ShowInputPrompt("Check # (Top Right)?")
    End Sub

    Private Sub FinishFlow()
        Me.DialogResult = DialogResult.OK
        Me.Close()
    End Sub

    Private Sub btnCancel_Click(sender As Object, e As EventArgs)
        Me.DialogResult = DialogResult.Cancel
        Me.Close()
    End Sub

    Private Sub FormAddChecks_KeyDown(sender As Object, e As KeyEventArgs) Handles MyBase.KeyDown
        If e.KeyCode = Keys.Escape Then
            btnCancel.PerformClick()
            e.Handled = True
            Return
        End If

        If currentStep = 5 Then
            If e.KeyCode = Keys.C Then
                ShowCheckNumberPrompt()
                e.Handled = True
                e.SuppressKeyPress = True
                Return
            End If

            If e.KeyCode = Keys.P Then
                MessageBox.Show("Print list of invoices is not implemented yet.",
                                "Add A Check",
                                MessageBoxButtons.OK,
                                MessageBoxIcon.Information)
                e.Handled = True
                e.SuppressKeyPress = True
                Return
            End If

            If e.KeyCode = Keys.Q Then
                btnCancel.PerformClick()
                e.Handled = True
                e.SuppressKeyPress = True
                Return
            End If

            Return
        End If

        If e.KeyCode = Keys.Enter Then
            ProcessStep()
            e.Handled = True
            e.SuppressKeyPress = True
            Return
        End If
    End Sub

    Private Sub ProcessStep()
        Dim value As String
        value = txtInput.Text.Trim()

        If currentStep = 1 Then
            If UCase(value) = "Q" Then
                btnCancel.PerformClick()
                Return
            End If

            If value = "" Then
                txtInput.Focus()
                Return
            End If

            enteredCompany = value
            ShowAmountPrompt()
            Return
        End If

        If currentStep = 2 Then
            If value = "" Then
                txtInput.Focus()
                Return
            End If

            enteredAmount = value
            ShowLowestInvoicePrompt()
            Return
        End If

        If currentStep = 3 Then
            If value = "" Then
                txtInput.Focus()
                Return
            End If

            enteredLowestInvoice = value

            If value = "-1" Then
                ShowReviewPrompt()
                Return
            End If

            ShowHighestInvoicePrompt()
            Return
        End If

        If currentStep = 4 Then
            If value = "" Then
                txtInput.Focus()
                Return
            End If

            enteredHighestInvoice = value
            ShowReviewPrompt()
            Return
        End If

        If currentStep = 6 Then
            If value = "" Then
                txtInput.Focus()
                Return
            End If

            enteredCheckNumber = value
            FinishFlow()
            Return
        End If
    End Sub

End Class
