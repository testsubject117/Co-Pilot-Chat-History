# Step 2: create `CheckInvReader.vb`

```vbnet name=CheckInvReader.vb
Option Strict On
Option Explicit On

Imports System
Imports System.Collections.Generic
Imports System.Globalization
Imports System.IO
Imports Microsoft.VisualBasic

Public NotInheritable Class CheckInvReader
    Private Sub New()
    End Sub

    Public Shared Function ReadAll(path As String) As List(Of CheckInvRecord)
        Dim results As New List(Of CheckInvRecord)()

        If Not File.Exists(path) Then
            Return results
        End If

        Using sr As New StreamReader(path)
            While Not sr.EndOfStream
                Dim customer As String = ReadValue(sr)
                If customer Is Nothing Then Exit While

                Dim checkNumber As String = ReadValue(sr)
                Dim reference As String = ReadValue(sr)
                Dim dateText As String = ReadValue(sr)
                Dim amountText As String = ReadValue(sr)
                Dim countText As String = ReadValue(sr)

                Dim record As New CheckInvRecord()
                record.Customer = customer
                record.CheckNumber = checkNumber
                record.Reference = reference
                record.DateText = dateText

                Dim amt As Decimal = 0D
                Decimal.TryParse(amountText, NumberStyles.Any, CultureInfo.InvariantCulture, amt)
                record.CheckAmount = amt

                Dim count As Integer = 0
                Integer.TryParse(countText, NumberStyles.Any, CultureInfo.InvariantCulture, count)

                For i As Integer = 1 To count
                    Dim invText As String = ReadValue(sr)
                    Dim invNo As Integer = 0
                    Integer.TryParse(invText, NumberStyles.Any, CultureInfo.InvariantCulture, invNo)
                    record.InvoiceNumbers.Add(invNo)
                Next

                results.Add(record)
            End While
        End Using

        Return results
    End Function

    Private Shared Function ReadValue(sr As StreamReader) As String
        If sr.EndOfStream Then Return Nothing

        Dim line As String = sr.ReadLine()
        If line Is Nothing Then Return Nothing

        line = line.Trim()

        If line.Length >= 2 AndAlso line.StartsWith("""", StringComparison.Ordinal) AndAlso line.EndsWith("""", StringComparison.Ordinal) Then
            Return line.Substring(1, line.Length - 2)
        End If

        Return line
    End Function
End Class
```

# Step 3: create `CheckInvWriter.vb`

```vbnet name=CheckInvWriter.vb
Option Strict On
Option Explicit On

Imports System
Imports System.Collections.Generic
Imports System.Globalization
Imports System.IO
Imports System.Text

Public NotInheritable Class CheckInvWriter
    Private Sub New()
    End Sub

    Public Shared Sub WriteAll(path As String, records As IEnumerable(Of CheckInvRecord))
        Dim lines As New List(Of String)()

        For Each record In records
            lines.AddRange(ToLines(record))
        Next

        File.WriteAllLines(path, lines, Encoding.ASCII)
    End Sub

    Public Shared Sub Append(path As String, record As CheckInvRecord)
        Dim lines = ToLines(record)

        Using sw As New StreamWriter(path, append:=True, encoding:=Encoding.ASCII)
            For Each line In lines
                sw.WriteLine(line)
            Next
        End Using
    End Sub

    Private Shared Function ToLines(record As CheckInvRecord) As List(Of String)
        Dim lines As New List(Of String)()

        lines.Add(Quote(record.Customer))
        lines.Add(Quote(record.CheckNumber))
        lines.Add(Quote(record.Reference))
        lines.Add(Quote(record.DateText))
        lines.Add(record.CheckAmount.ToString(CultureInfo.InvariantCulture))
        lines.Add(record.InvoiceNumbers.Count.ToString(CultureInfo.InvariantCulture))

        For Each invNo In record.InvoiceNumbers
            lines.Add(invNo.ToString(CultureInfo.InvariantCulture))
        Next

        Return lines
    End Function

    Private Shared Function Quote(value As String) As String
        If value Is Nothing Then value = ""
        value = value.Replace("""", "")
        Return """" & value & """"
    End Function
End Class
```
