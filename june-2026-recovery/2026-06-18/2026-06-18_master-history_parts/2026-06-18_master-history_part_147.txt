## 2) `RolodexAreaCodeService.vb`

This version supports:

- `A` = all grouped listings
- `CA` = grouped region lookup
- `213` = area code lookup
- `Q` can be handled by the form/menu
- DOS-style grouped text output
- `GetAllPages()` returns `List(Of String)` as your form expects

```vb name=RolodexAreaCodeService.vb
Imports System
Imports System.Collections.Generic
Imports System.IO
Imports System.Linq
Imports System.Text

Public Class RolodexAreaCodeService

    Private Shared ReadOnly DataPath As String = "\\invoice\mainmenu\data\phone\areacodes.csv"
    Private ReadOnly _entries As List(Of AreaCodeEntry)

    Public Sub New()
        _entries = LoadEntries()
    End Sub

    Public Function DataFileExists() As Boolean
        Return File.Exists(DataPath)
    End Function

    Public Function GetDataFilePath() As String
        Return DataPath
    End Function

    Public Function Search(input As String) As List(Of AreaCodeEntry)
        Dim results As New List(Of AreaCodeEntry)

        If String.IsNullOrWhiteSpace(input) Then
            Return results
        End If

        Dim trimmed = input.Trim()

        If String.Equals(trimmed, "A", StringComparison.OrdinalIgnoreCase) Then
            Return _entries.
                OrderBy(Function(x) x.RegionCode).
                ThenBy(Function(x) x.AreaCode).
                ThenBy(Function(x) x.Location).
                ToList()
        End If

        If trimmed.Length = 2 AndAlso trimmed.All(Function(c) Char.IsLetter(c)) Then
            Dim regionCode = trimmed.ToUpperInvariant()

            results = _entries.
                Where(Function(x) String.Equals(x.RegionCode, regionCode, StringComparison.OrdinalIgnoreCase)).
                OrderBy(Function(x) x.AreaCode).
                ThenBy(Function(x) x.Location).
                ToList()

            Return results
        End If

        Dim digitsOnly = New String(trimmed.Where(Function(c) Char.IsDigit(c)).ToArray())

        If digitsOnly.Length > 0 Then
            results = _entries.
                Where(Function(x) Not String.IsNullOrWhiteSpace(x.AreaCode) AndAlso
                                  x.AreaCode.StartsWith(digitsOnly, StringComparison.OrdinalIgnoreCase)).
                OrderBy(Function(x) x.AreaCode).
                ThenBy(Function(x) x.RegionCode).
                ThenBy(Function(x) x.Location).
                ToList()

            Return results
        End If

        Return results
    End Function

    Public Function FormatGroupedResults(entries As List(Of AreaCodeEntry)) As String
        Dim sb As New StringBuilder()

        If entries Is Nothing OrElse entries.Count = 0 Then
            sb.AppendLine("No matches found.")
            Return sb.ToString()
        End If

        Dim grouped = entries.
            GroupBy(Function(x) New With {Key .RegionCode = x.RegionCode, Key .RegionName = x.RegionName}).
            OrderBy(Function(g) g.Key.RegionCode)

        For Each grp In grouped
            sb.AppendLine(grp.Key.RegionCode & "  " & grp.Key.RegionName & " area code(s):")

            For Each item In grp.OrderBy(Function(x) x.AreaCode).ThenBy(Function(x) x.Location)
                sb.AppendLine("   " & item.AreaCode.PadRight(4) & " " & item.Location)
            Next

            sb.AppendLine()
        Next

        Return sb.ToString().TrimEnd()
    End Function

    Public Function GetAllPages(Optional linesPerPage As Integer = 35) As List(Of String)
        Dim pages As New List(Of String)

        If linesPerPage <= 0 Then
            linesPerPage = 35
        End If

        Dim allText = FormatGroupedResults(
            _entries.
                OrderBy(Function(x) x.RegionCode).
                ThenBy(Function(x) x.AreaCode).
                ThenBy(Function(x) x.Location).
                ToList())

        If String.IsNullOrWhiteSpace(allText) Then
            Return pages
        End If

        Dim lines = allText.Replace(vbCrLf, vbLf).Split(ControlChars.Lf)
        Dim current As New List(Of String)

        For Each line In lines
            current.Add(line)

            If current.Count >= linesPerPage Then
                pages.Add(String.Join(Environment.NewLine, current))
                current.Clear()
            End If
        Next

        If current.Count > 0 Then
            pages.Add(String.Join(Environment.NewLine, current))
        End If

        Return pages
    End Function

    Private Function LoadEntries() As List(Of AreaCodeEntry)
        Dim results As New List(Of AreaCodeEntry)

        If Not File.Exists(DataPath) Then
            Return results
        End If

        Dim isFirstLine As Boolean = True

        For Each rawLine In File.ReadAllLines(DataPath)
            If String.IsNullOrWhiteSpace(rawLine) Then Continue For

            If isFirstLine Then
                isFirstLine = False
                If rawLine.Trim().StartsWith("RegionCode", StringComparison.OrdinalIgnoreCase) Then
                    Continue For
                End If
            End If

            Dim parts = rawLine.Split("|"c)
            If parts.Length < 4 Then Continue For

            Dim entry As New AreaCodeEntry With {
                .RegionCode = parts(0).Trim().ToUpperInvariant(),
                .RegionName = parts(1).Trim(),
                .AreaCode = parts(2).Trim(),
                .Location = parts(3).Trim()
            }

            If String.IsNullOrWhiteSpace(entry.RegionCode) Then Continue For
            If String.IsNullOrWhiteSpace(entry.RegionName) Then Continue For
            If String.IsNullOrWhiteSpace(entry.AreaCode) Then Continue For
            If String.IsNullOrWhiteSpace(entry.Location) Then Continue For

            results.Add(entry)
        Next

        Return results
    End Function
End Class
```

Private Sub OpenAreaCodes()
    Dim svc As New RolodexAreaCodeService()

    If Not svc.DataFileExists() Then
        MessageBox.Show("Area code data file not found:" & Environment.NewLine & svc.GetDataFilePath(),
                        "Area Code Finder",
                        MessageBoxButtons.OK,
                        MessageBoxIcon.Warning)
        Return
    End If

    Dim prompt As String =
        "Enter Area Code to find city or 2 letter abbreviation of state to find area code" & Environment.NewLine &
        "Example: California = CA" & Environment.NewLine & Environment.NewLine &
        "[A = All] [Q = Quit] ?"

    Dim userInput As String = InputBox(prompt, "LOOK UP AREA CODES")

    If String.IsNullOrWhiteSpace(userInput) Then
        Return
    End If

    Dim trimmed = userInput.Trim()

    If String.Equals(trimmed, "Q", StringComparison.OrdinalIgnoreCase) Then
        Return
    End If

    Using frm As New FrmPagedTextViewer()
        If String.Equals(trimmed, "A", StringComparison.OrdinalIgnoreCase) Then
            Dim pages = svc.GetAllPages()

            If pages Is Nothing OrElse pages.Count = 0 Then
                MessageBox.Show("No area code data found.",
                                "Area Code Finder",
                                MessageBoxButtons.OK,
                                MessageBoxIcon.Information)
                Return
            End If

            frm.SetPages(pages)
            frm.ShowDialog(Me)
            Return
        End If

        Dim results = svc.Search(trimmed)

        If results Is Nothing OrElse results.Count = 0 Then
            MessageBox.Show("No matches found.",
                            "Area Code Finder",
                            MessageBoxButtons.OK,
                            MessageBoxIcon.Information)
            Return
        End If

        Dim formatted As String = svc.FormatGroupedResults(results)
        frm.SetPages(New List(Of String) From {formatted})
        frm.ShowDialog(Me)
    End Using
End Sub
