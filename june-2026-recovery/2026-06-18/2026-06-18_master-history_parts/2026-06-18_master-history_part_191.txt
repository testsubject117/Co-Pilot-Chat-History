        }

        Me.Controls.Add(txtScreen)
        Me.Controls.Add(lblTitle)

        AddHandler Me.Load, AddressOf FrmRolodexPrompt_Load
        AddHandler Me.Shown, AddressOf FrmRolodexPrompt_Shown
        AddHandler Me.KeyPress, AddressOf FrmRolodexPrompt_KeyPress
        AddHandler Me.KeyDown, AddressOf FrmRolodexPrompt_KeyDown
        AddHandler txtScreen.Click, AddressOf RefocusScreen
        AddHandler Me.Click, AddressOf RefocusScreen
    End Sub

    Private Sub FrmRolodexPrompt_Load(sender As Object, e As EventArgs)
        lblTitle.Text = _engine.Title
        RefreshScreen()
    End Sub

    Private Sub FrmRolodexPrompt_Shown(sender As Object, e As EventArgs)
        RefocusScreen(Nothing, EventArgs.Empty)
    End Sub

    Private Sub RefocusScreen(sender As Object, e As EventArgs)
        Me.Activate()
        txtScreen.Focus()
    End Sub

    Private Sub FrmRolodexPrompt_KeyPress(sender As Object, e As KeyPressEventArgs)
        If Char.IsControl(e.KeyChar) Then Return

        _currentInput &= e.KeyChar
        RefreshScreen()
        e.Handled = True
    End Sub

    Private Sub FrmRolodexPrompt_KeyDown(sender As Object, e As KeyEventArgs)
        If e.KeyCode = Keys.Back Then
            If _currentInput.Length > 0 Then
                _currentInput = _currentInput.Substring(0, _currentInput.Length - 1)
                RefreshScreen()
            End If
            e.SuppressKeyPress = True
            Return
        End If

        If e.KeyCode = Keys.Enter Then
            e.SuppressKeyPress = True

            Dim submitted = _currentInput
            _currentInput = ""

            _engine.SubmitInput(submitted)

            If _engine.IsComplete() Then
                HandleCompletedFlow()
                Return
            End If

            RefreshScreen()
            Return
        End If

        If e.KeyCode = Keys.Escape Then
            e.SuppressKeyPress = True
            Me.DialogResult = DialogResult.Cancel
            Me.Close()
        End If
    End Sub

        Dim lines As New List(Of String)
        lines.AddRange(_engine.Transcript)

        Dim prompt = _engine.GetCurrentPrompt()
        If Not String.IsNullOrWhiteSpace(prompt) Then
            lines.Add(AppendInlineInput(prompt, _currentInput))
        End If

        txtScreen.Text = String.Join(Environment.NewLine & Environment.NewLine, lines)
        txtScreen.SelectionStart = txtScreen.TextLength
        txtScreen.ScrollToCaret()
    End Sub

    Private Function AppendInlineInput(prompt As String, input As String) As String
        Dim parts = prompt.Split({Environment.NewLine}, StringSplitOptions.None)

        If parts.Length = 0 Then
            Return input
        End If

        parts(parts.Length - 1) &= " " & input

        Return String.Join(Environment.NewLine, parts)
    End Function

    Private Sub HandleCompletedFlow()
        RefreshScreen()

        If RequiresAcceptancePrompt() Then
            ShowReviewAndCloseIfAccepted()
        Else
            Me.DialogResult = DialogResult.OK
            Me.Close()
        End If
    End Sub

    Private Function RequiresAcceptancePrompt() As Boolean
        Dim title = _engine.Title.ToUpperInvariant()

        If title.Contains("ADD A PERSON") Then Return True
        If title.Contains("MODIFY A PERSON") Then Return True

        Return False
    End Function

    Private Sub ShowReviewAndCloseIfAccepted()
        Dim reviewLines = _engine.GetReviewLines()
        Dim reviewText = String.Join(Environment.NewLine, reviewLines)

        Dim result = MessageBox.Show(
            reviewText & Environment.NewLine & Environment.NewLine & "Accept this record?",
            "Rolodex",
            MessageBoxButtons.YesNo,
            MessageBoxIcon.Question
        )

        If result = DialogResult.Yes Then
            Me.DialogResult = DialogResult.OK
        Else
            Me.DialogResult = DialogResult.Cancel
        End If

        Me.Close()
    End Sub
End Class

## Copilot

Usually, a GW-BASIC Rolodex would store data in **disk files**, not inside the `.BAS` program itself. GW-BASIC supported two main styles: **sequential files** (written/read in order with `PRINT#`/`WRITE#` and `INPUT#`/`LINE INPUT#`) and **random-access files** (fixed-length records using `OPEN ... FOR RANDOM`, `FIELD`, `GET`, and `PUT`). ([hwiegman.home.xs4all.nl](https://hwiegman.home.xs4all.nl/gw-man/Chapter%205.html?utm_source=openai))

For something like a Rolodex that needs **add / find / modify / delete** by record, the classic pattern was often a **random-access file**: the BASIC program defined a fixed record layout with `FIELD`, loaded one record with `GET`, changed fields in a buffer, then saved it back with `PUT`. GW-BASICâ€™s docs explicitly describe random files as suitable for keyed record storage and show that records are addressed by record number rather than rereading the whole file. ([www-user.tu-chemnitz.de](https://www-user.tu-chemnitz.de/~heha/hs/chm/gwbasic.chm/Chapter5.html?utm_source=openai))

A simpler app could also use a **sequential text file**, where each contact is written one after another and the program scans from the top to find a match. But sequential files are less convenient for in-place edits, because updating one record often means rereading and rewriting much or all of the file. GW-BASIC documentation contrasts this with random-access storage, noting sequential data is stored â€œone item after anotherâ€ and random-access files are better when you need record-oriented updates. ([hwiegman.home.xs4all.nl](https://hwiegman.home.xs4all.nl/gw-man/Chapter%205.html?utm_source=openai))

So the likely answer is: **the GW-BASIC app probably stored Rolodex entries in a separate data file on disk, most likely a random-access flat file if it supported editing efficiently, or a sequential flat file if it was simpler.** The `.BAS` file usually contained the **logic** to read and write that file, not the live database itself. ([hwiegman.home.xs4all.nl](https://hwiegman.home.xs4all.nl/gw-man/Chapter%205.html?utm_source=openai))

The fastest way to know for sure in your app is to inspect the Rolodex BASIC source and look for file statements such as:
- `OPEN "..." FOR RANDOM`
- `FIELD`
- `GET` / `PUT`
- or `OPEN "..." FOR INPUT/OUTPUT`, `WRITE#`, `INPUT#`, `LINE INPUT#`  
Those lines will tell you the exact filename and storage style the DOS app used. ([www-user.tu-chemnitz.de](https://www-user.tu-chemnitz.de/~heha/hs/chm/gwbasic.chm/Chapter5.html?utm_source=openai))

## Bottom line
The Rolodex data was **not stored inside `PHONE.BAS` itself**.  
It was stored in a set of **per-letter sequential files** in the `PHONE` directory:

- `A.LST`
- `B.LST`
- ...
- `Z.LST`
- and also numeric starters:
- `0.LST` through `9.LST`

The program chooses the file by the **first character of the name** using `LEFT$(N$,1)+".LST"`. For Add, it appends a record to that file; for search/delete/modify, it opens the same per-letter file and scans each record with `INPUT#`. ([hwiegman.home.xs4all.nl](https://hwiegman.home.xs4all.nl/gw-man/INPUTF.html?utm_source=openai))

## Exact evidence from your code
### Add writes to `<first-letter>.LST`
In line 850 and again at 990:

- `Q$=LEFT$(N$,1)+".LST"`
- `OPEN Q$ FOR INPUT AS 1`
- later:
- `OPEN Q$ FOR APPEND LOCK READ WRITE AS 1:WRITE#1,N$,A$,C$,S$,Z,A,P,M$:CLOSE 1`

That means a name like `PLUTO` is stored in:
- `P.LST`

and a name like `TEST` is stored in:
- `T.LST`

Because it uses `WRITE#`, each record is a sequential line of 8 fields:
1. name
2. address
3. city
4. state
5. zip
6. area code
7. phone number
8. misc

GW-BASIC `WRITE#` writes comma-delimited records and quotes strings so they can be read back reliably with `INPUT#`; your code uses exactly that pair. ([hwiegman.home.xs4all.nl](https://hwiegman.home.xs4all.nl/gw-man/WRITEF.html?utm_source=openai))

### Delete / Lookup / Modify read the same `<first-letter>.LST`
At lines 1100 and 1130:

- `Q$=LEFT$(N2$,1)+".LST":OPEN Q$ FOR INPUT AS 1`
- `INPUT#1,N$,A$,C$,S$,Z,A,P,M$`

So when the user searches `PLUTO`, DOS opens:
- `P.LST`

and scans records from that file. It does **not** search a single monolithic Rolodex database file. It uses **one file per starting letter/number**. ([hwiegman.home.xs4all.nl](https://hwiegman.home.xs4all.nl/gw-man/INPUTF.html?utm_source=openai))

### Delete / Modify rebuild the letter file using a temp file
At lines 1110, 1240â€“1270, and 1340:

- opens `<letter>.TMP` for output
- copies surviving records there
- then if changes were made:
  - `KILL LEFT$(N$,1)+".LST"`
  - `NAME LEFT$(N2$,1)+".TMP " AS LEFT$(N2$,1)+".LST"`

That confirms `.LST` files are the live database files, and `.TMP` files are temporary rebuild files during delete/modify. ([hwiegman.home.xs4all.nl](https://hwiegman.home.xs4all.nl/gw-man/INPUTF.html?utm_source=openai))
