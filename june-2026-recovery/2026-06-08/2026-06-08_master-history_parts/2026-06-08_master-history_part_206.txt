```vbnet
Private Function Unquote(value As String) As String
    If value Is Nothing Then
        Return ""
    End If

    Dim s As String = value

    ' Remove Unicode BOM if present (can appear at start of first line)
    s = s.Replace(ChrW(&HFEFF), "")

    ' Trim first (after BOM removal)
    s = s.Trim()

    ' Remove surrounding quotes
    If s.Length >= 2 AndAlso
       s.StartsWith("""", StringComparison.Ordinal) AndAlso
       s.EndsWith("""", StringComparison.Ordinal) Then

        s = s.Substring(1, s.Length - 2)
    End If

    ' Unescape doubled quotes
    s = s.Replace("""""", """")

    ' Strip leading/trailing control characters (DOS artifacts, etc.)
    ' Keep normal printable characters and spaces.
    s = StripEdgeControlChars(s)

    Return s.Trim()
End Function

Private Function StripEdgeControlChars(input As String) As String
    If String.IsNullOrEmpty(input) Then Return ""

    Dim startIdx As Integer = 0
    Dim endIdx As Integer = input.Length - 1

    While startIdx <= endIdx AndAlso Char.IsControl(input(startIdx))
        startIdx += 1
    End While

    While endIdx >= startIdx AndAlso Char.IsControl(input(endIdx))
        endIdx -= 1
    End While

    If startIdx = 0 AndAlso endIdx = input.Length - 1 Then
        Return input
    End If

    If startIdx > endIdx Then
        Return ""
    End If

    Return input.Substring(startIdx, (endIdx - startIdx) + 1)
End Function
```

That will remove the weird glyph if itâ€™s coming from a control/BOM character at the edge of the string (which matches what youâ€™re seeing: **first character of the first real displayed entry**).

        If panel Is Nothing Then Return

            panel.ClientSize.Width -
            panel.Padding.Left - panel.Padding.Right -

        If targetWidth < 150 Then targetWidth = 150

        For Each c As Control In panel.Controls
            Dim btn = TryCast(c, Button)
            Dim lbl = TryCast(c, Label)

            If btn IsNot Nothing Then
                btn.Width = targetWidth
            ElseIf lbl IsNot Nothing Then
                lbl.Width = targetWidth
            End If
        Next

    ' -------------------------------

    Private Sub EnsurePromptUi()
        If pnlPromptHost IsNot Nothing Then Return

        pnlPromptHost = New Panel() With {
            .Visible = False,
            .Height = 28,
            .Dock = DockStyle.Bottom,
            .BackColor = Color.Black,
            .Padding = New Padding(10, 3, 10, 3)
        }

        lblPrompt = New Label() With {
            .Dock = DockStyle.Fill,
            .BackColor = Color.Black,
            .ForeColor = Color.White,
            .TextAlign = ContentAlignment.MiddleLeft,
            .Font = New Font("Consolas", 12.0F, FontStyle.Regular, GraphicsUnit.Point),
            .AutoEllipsis = True
        }

        pnlPromptHost.Controls.Add(lblPrompt)
        Me.Controls.Add(pnlPromptHost)
        pnlPromptHost.BringToFront()

        ' Draw a 1px border around the prompt bar
        AddHandler pnlPromptHost.Paint, AddressOf pnlPromptHost_Paint
    End Sub

    Private Sub pnlPromptHost_Paint(sender As Object, e As PaintEventArgs)
        Dim p As Panel = DirectCast(sender, Panel)

        Using pen As New Pen(_promptBorderColor, 1.0F)
            ' Draw border inside bounds so it is fully visible
            Dim r As Rectangle = p.ClientRectangle
            r.Width -= 1
            r.Height -= 1
            e.Graphics.DrawRectangle(pen, r)
        End Using
    End Sub

    Private Sub ShowBottomPrompt(text As String, Optional borderColor As Color = Nothing)
        EnsurePromptUi()

        If borderColor = Nothing Then
            _promptBorderColor = Color.Yellow
        Else
            _promptBorderColor = borderColor
        End If

        lblPrompt.Text = text
        pnlPromptHost.Visible = True
        pnlPromptHost.BringToFront()
        pnlPromptHost.Invalidate()
    End Sub

    Private Sub HideBottomPrompt()
        If pnlPromptHost Is Nothing Then Return
        pnlPromptHost.Visible = False
        If lblPrompt IsNot Nothing Then lblPrompt.Text = ""
    End Sub

    Private Sub StartActualNamesPrompt()
        _isActualNamesPromptActive = True

        ' Border can be Color.White if you prefer; Yellow stands out more.
        ShowBottomPrompt("Push P to Print Out Actual Names, or Q to Quit", Color.Yellow)
    End Sub

    Private Sub StopActualNamesPrompt()
        _isActualNamesPromptActive = False
        HideBottomPrompt()
    End Sub

                            "Actual Customer Names",
                            MessageBoxButtons.OK,
                            MessageBoxIcon.Error)
        End Try
    End Sub
