    Private Sub ApplyFilters()
        Dim cust As String = txtCustomer.Text.Trim()
        Dim chk As String = txtCheck.Text.Trim()

        Dim month As String = If(TryCast(cmbMonth.SelectedItem, String), "").Trim()

    Private Function GetInvoiceNumbersForEntry(entry As LedgerEntry) As List(Of Long)
        If entry Is Nothing Then
            Return New List(Of Long)()
        End If

        Dim keys As List(Of String) = BuildCandidateKeys(entry.Customer, entry.CheckNumber)

        Dim i As Integer
        For i = 0 To keys.Count - 1
            Dim key As String = keys(i)
            Dim invoiceNumbers As List(Of Long) = Nothing

            If _invIndex.TryGetValue(key, invoiceNumbers) AndAlso invoiceNumbers IsNot Nothing AndAlso invoiceNumbers.Count > 0 Then
                Return New List(Of Long)(invoiceNumbers)
            End If
        Next

        Return New List(Of Long)()
    End Function

    Private Sub StartLoadCheckInvIndex()
        lblInvoiceStatus.Text = "Loading CHECK.INV index..."
        Dim path As String = LegacyDataPaths.CheckInv

        Task.Run(
            Sub()
                Dim idx As New Dictionary(Of String, List(Of Long))(StringComparer.OrdinalIgnoreCase)

                Try
                    If System.IO.File.Exists(path) Then
                        Dim blocks = CheckInvReader.ReadAll(path)

                        Dim bi As Integer
                        For bi = 0 To blocks.Count - 1
                            Dim b = blocks(bi)
                            Dim keys As List(Of String) = BuildCandidateKeys(b.CustomerCode, b.CheckNumber)
                            Dim parsedInvoices As New List(Of Long)()

                            If b.Invoices IsNot Nothing Then
                                Dim ii As Integer
                                For ii = 0 To b.Invoices.Count - 1
                                    Dim invStr As String = b.Invoices(ii)
                                    Dim invNo As Long
                                    If Long.TryParse(SafeTrim(invStr), invNo) Then
                                        If Not parsedInvoices.Contains(invNo) Then
                                            parsedInvoices.Add(invNo)
                                        End If
                                    End If
                                Next
                            End If

                            Dim ki As Integer
                            For ki = 0 To keys.Count - 1
                                Dim key As String = keys(ki)
                                If key = "" Then Continue For

                                Dim lst As List(Of Long) = Nothing
                                If Not idx.TryGetValue(key, lst) Then
                                    lst = New List(Of Long)()
                                    idx(key) = lst
                                End If

                                Dim pi As Integer
                                For pi = 0 To parsedInvoices.Count - 1
                                    Dim invNo As Long = parsedInvoices(pi)
                                    If Not lst.Contains(invNo) Then
                                        lst.Add(invNo)
                                    End If
                                Next
                            Next
                        Next
                    End If
                Catch
                End Try

                Try
                    If Not Me.IsDisposed Then
                        Me.Invoke(
                            Sub()
                                _invIndex = idx
                                _invIndexReady = True
                                ApplyFilters()
                                OnMainGridSelectionChanged()
                            End Sub)
                    End If
                Catch __e As ObjectDisposedException
                End Try
            End Sub)
    End Sub

    Private Shared Function BuildCandidateKeys(code As String, checkNum As String) As List(Of String)
        Dim results As New List(Of String)()

        Dim codeTrimmed As String = SafeTrim(code)
        Dim checkTrimmed As String = SafeTrim(checkNum)

        Dim codeUpper As String = codeTrimmed.ToUpperInvariant()
        Dim checkUpper As String = checkTrimmed.ToUpperInvariant()

        Dim codeCompact As String = codeUpper.Replace(" ", "")
        Dim checkCompact As String = checkUpper.Replace(" ", "")

        AddUnique(results, codeUpper & "|" & checkUpper)
        AddUnique(results, codeTrimmed & "|" & checkTrimmed)
        AddUnique(results, codeCompact & "|" & checkUpper)
        AddUnique(results, codeUpper & "|" & checkCompact)
        AddUnique(results, codeCompact & "|" & checkCompact)

        Return results
    End Function

    Private Shared Sub AddUnique(list As List(Of String), value As String)
        If list Is Nothing Then Return
        If String.IsNullOrWhiteSpace(value) Then Return

        Dim i As Integer
        For i = 0 To list.Count - 1
            If String.Equals(list(i), value, StringComparison.OrdinalIgnoreCase) Then
                Return
            End If
        Next

        list.Add(value)
    End Sub

    Private Shared Function SafeTrim(value As String) As String
        Return If(value, "").Trim()
    End Function

End Class

## Copilot

Yes â€” this is enough to explain the legacy behavior.

## Direct answer
The DOS app **did not** use `CHECK.INV` invoice numbers to look up amount/status in `INVOICE.CHK` during **View Checks**.

Thatâ€™s the key.

From `LEDGER.ASC`:

### In `View Checks` (`460` block)
It reads only:

- `LEDGER.CUR`

and displays:
- customer
- date
- check #
- reference
- â€œDifference Between Total of Invoices & Check Amountâ€
- amount

It does **not** read `CHECK.INV` there.  
It does **not** read `INVOICE.CHK` there.  
So the original View Checks screen was just a ledger summary viewer.

---

## Where `CHECK.INV` was used in DOS
`CHECK.INV` was used for two lookup functions:

### Find invoices paid by a check
At `2870+`
- opens `CHECK.INV`
- reads:
  - customer
  - check #
  - reference
  - date
  - amount
  - count
- then reads invoice numbers
- displays invoice list

### Find the check that paid an invoice
At `3060+`
- opens `CHECK.INV`
- scans blocks
- checks whether a given invoice number exists in the block
- then displays the related check

So in the DOS app, **`CHECK.INV` was the invoice-list bridge file**, not the source for invoice-detail enrichment inside View Checks.

---

## Where `INVOICE.CHK` was used
`INVOICE.CHK` was used when:

### adding checks
At `790+`
- invoice numbers are entered manually
- file is read by random access:
  - `IFIND = INT(INUM1 - 75000!)`
  - `GET #1, IFIND`
- then amount/company/status are validated there

### auto-scanning unpaid invoices
At `3500+`
- it scans `INVOICE.CHK`
- builds invoice candidates by company and unpaid flag
- then uses:
  - `HOLD(CT)=X`
  - displayed invoice number = `X + 75000`

### deleting a check / resetting invoices to unpaid
At `3910+`
- it takes invoice numbers from `CHECK.INV`
- converts them:
  - `HOLD = INT(HOLD(X) - 75000!)`
- then directly updates `INVOICE.CHK`

So this tells us something crucial:

# In the original DOS system, the invoice numbers stored in `CHECK.INV` were expected to be direct invoice numbers in the same `+75000` scheme used by `INVOICE.CHK`.

---

## But your live data contradicts that
Your live debug proved:

- `CHECK.INV` contains numbers like `190815`
- `INVOICE.CHK` near matching customer rows decodes invoice numbers like `231630`

That means one of these is true:

1. your current `INVOICE.CHK` is not in the same numbering basis as the old DOS code expected
2. your MBF decode of the invoice-number field is wrong even though amount/company/flag decode looks right
3. the invoice number field in `INVOICE.CHK` should **not** be trusted for validation, and only the **record position** matters

And now, because of the DOS source, I think **#3 is the likely answer**.

---

# The actual legacy lookup rule
From the DOS code:

```basic
IFIND=INT(INUM1-75000!)
LOCK#1,IFIND:GET#1,IFIND
IF IFIND<>CVS(INUM2$) THEN ...
```

Notice what it compares:

- `IFIND`
- to `CVS(INUM2$)`

Not:
- invoice number `190815`
- but record index `190815 - 75000 = 115815`

That means the `INUM2$` field inside `INVOICE.CHK` probably stores the **record number / internal index**, not the full displayed invoice number.

That perfectly explains your debug mismatch.

Example:

- invoice number: `190815`
- IFIND = `190815 - 75000 = 115815`

Your debug decoded something around:
- `231630`

That is almost exactly:
- `115815 * 2 = 231630`

That is not random.

---

# This is the breakthrough
Your MBF invoice-number decode is off by a factor of **2** relative to the DOS `CVS` meaning being used here.

So the stored `INUM2$` value is representing something that, when decoded by your current MBF function, appears doubled.

That means:

## either:
- the MBF decoder is not actually correct for GW-BASIC `CVS` here

## or more practically:
- for validation, you should compare the decoded stored value to:
  - `2 * IFIND`
instead of
  - the requested invoice number

And record lookup should still use:
- `record = invoice - 75000`

That matches the DOS source exactly.

---

# What this means for the fix
The scan-every-record approach was wrong for this file.
The correct approach is to go back to direct random-access lookup, but validate using the DOS semantics.

Specifically:

- requested invoice = `190815`
- record number = `190815 - 75000 = 115815`
- read record `115815`
- decode `INUM2$`
- if decoded value â‰ˆ `recordNumber * 2`, accept it
- decode amount/company/flag from same record

That matches your debug exactly:
- record `115815`
- decoded invoice raw `231630`
- `115815 * 2 = 231630`

Perfect match.

So the original system was working by **record number**, not by full invoice number stored in that field the way we assumed.

# Full replacement: `AMiOffice/InvoiceChkReader.vb`

Use this version. It restores direct-record lookup and validates against the actual DOS behavior revealed by both the source and your debug.

```vbnet
Option Strict On
Option Explicit On

Imports System.IO
Imports System.Text

''' <summary>
''' Reads random-access binary records from INVOICE.CHK.
''' Legacy GW-BASIC code uses:
'''   IFIND = INT(InvoiceNumber - 75000!)
'''   GET #1, IFIND
''' and validates:
'''   IF IFIND <> CVS(INUM2$) THEN ...
'''
''' In the live data, the decoded INUM2$ value appears to be stored on a scale
''' that is effectively 2 * recordNumber, so validation must account for that.
''' </summary>
Public NotInheritable Class InvoiceChkReader
    Private Sub New()
    End Sub

    Private Const RecordSize As Integer = 26
    Private Const InvoiceBase As Long = 75000L

    Public Shared Function DecodeMbfSingle(buf() As Byte, offset As Integer) As Single
        If buf Is Nothing OrElse offset < 0 OrElse offset + 4 > buf.Length Then Return 0.0F

        Dim b0 As Byte = buf(offset)
        Dim b1 As Byte = buf(offset + 1)
        Dim b2 As Byte = buf(offset + 2)
        Dim b3 As Byte = buf(offset + 3)

        If b3 = 0 Then Return 0.0F

        Dim signBit As UInteger = (CUInt(b2) And &H80UI) << 24
        Dim ieeeExp As UInteger = CUInt(b3) - 1UI
        Dim mantissa As UInteger = ((CUInt(b2) And &H7FUI) << 16) Or
                                   (CUInt(b1) << 8) Or
                                   CUInt(b0)

        Dim bits As UInteger = signBit Or (ieeExp << 23) Or mantissa

        Dim ieee4(3) As Byte
        ieee4(0) = CByte(bits And &HFFUI)
        ieee4(1) = CByte((bits >> 8) And &HFFUI)
        ieee4(2) = CByte((bits >> 16) And &HFFUI)
        ieee4(3) = CByte((bits >> 24) And &HFFUI)

        Dim result As Single = BitConverter.ToSingle(ieee4, 0)
        If Single.IsNaN(result) OrElse Single.IsInfinity(result) Then Return 0.0F
        Return result
    End Function

    Public Shared Function ReadRecord(path As String, invoiceNumber As Long) As InvoiceChkRecord
        Dim rec As New InvoiceChkRecord() With {.InvoiceNumber = invoiceNumber}

        If invoiceNumber <= 0 Then Return rec
        If Not File.Exists(path) Then Return rec

        Dim recordNumber As Long = invoiceNumber - InvoiceBase
        If recordNumber < 1 Then Return rec

        Try
            Using fs As New FileStream(path, FileMode.Open, FileAccess.Read, FileShare.ReadWrite)
                Dim recordOffset As Long = (recordNumber - 1) * RecordSize
                If recordOffset < 0 OrElse recordOffset + RecordSize > fs.Length Then
                    Return rec
                End If

                Dim buf(RecordSize - 1) As Byte
                fs.Seek(recordOffset, SeekOrigin.Begin)
                Dim bytesRead As Integer = fs.Read(buf, 0, RecordSize)
                If bytesRead < RecordSize Then
                    Return rec
                End If

                Dim storedIndexRaw As Single = DecodeMbfSingle(buf, 0)
                Dim storedAmountRaw As Single = DecodeMbfSingle(buf, 9)

                Dim storedIndexRounded As Long = CLng(Math.Round(CDbl(storedIndexRaw), MidpointRounding.AwayFromZero))
                Dim expectedStoredIndex As Long = recordNumber * 2L

                ' DOS logic validates the record slot, not the displayed invoice number.
                ' In live data, the decoded stored value tracks as 2 * record number.
                If storedIndexRounded <> expectedStoredIndex Then
                    Return rec
                End If

                rec.Amount = CDec(storedAmountRaw)
                rec.CompanyCode = Encoding.ASCII.GetString(buf, 17, 8).TrimEnd(ChrW(0), " "c)

                Dim flagByte As Byte = buf(25)
                If flagByte <> 0 AndAlso flagByte <> 32 Then
                    rec.Flag = ChrW(flagByte).ToString().ToUpperInvariant()
                Else
                    rec.Flag = ""
                End If

                rec.IsFound = True
            End Using
        Catch
            ' Treat any I/O/format problem as not found.
        End Try

        Return rec
    End Function
End Class

''' <summary>Decoded record from INVOICE.CHK.</summary>
Public Class InvoiceChkRecord
    Public Property InvoiceNumber As Long
    Public Property Amount As Decimal
    Public Property CompanyCode As String = ""
    Public Property Flag As String = ""
    Public Property IsFound As Boolean = False

    Public ReadOnly Property FlagDescription As String
        Get
            Select Case Flag
                Case "J" : Return "Open"
                Case "P" : Return "Paid"
                Case "C" : Return "Corrected Cert"
                Case "V" : Return "Void"
                Case "E" : Return "Erased"
                Case Else : Return If(Flag, "")
            End Select
        End Get
    End Property
End Class

''' <summary>
''' Flat row bound to the invoice-details DataGridView inside FormLedgerView.
''' </summary>
Public Class InvoiceDetailRow
    Public Property InvoiceNumber As Long
    Public Property Amount As String = ""
    Public Property CompanyCode As String = ""
    Public Property Status As String = ""
End Class
```
