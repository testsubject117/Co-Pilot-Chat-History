    Private Sub PositionInlineTextbox()
        If Not txtInline.Visible Then Return
        If String.IsNullOrWhiteSpace(_activePromptText) Then Return

        Dim fullText As String = If(lblScreen.Text, "")
        Dim idx As Integer = fullText.LastIndexOf(_activePromptText, StringComparison.OrdinalIgnoreCase)
        If idx < 0 Then
            ' Fallback: top-left
            txtInline.Left = 10
            txtInline.Top = 10
            txtInline.Width = 400
            Return
        End If

        ' Determine the line number containing the prompt
        Dim before As String = fullText.Substring(0, idx)
        Dim linesBefore As String() = before.Split(New String() {vbCrLf}, StringSplitOptions.None)
        Dim lineIndex As Integer = linesBefore.Length - 1

        ' For monospaced font: we can compute x by prompt length.
        ' We'll put the textbox after: prompt + space
        Dim col As Integer = _activePromptText.Length + 1

        Dim charW As Integer = TextRenderer.MeasureText("W", lblScreen.Font, New Size(Integer.MaxValue, Integer.MaxValue), TextFormatFlags.NoPadding).Width
        Dim lineH As Integer = TextRenderer.MeasureText("W", lblScreen.Font, New Size(Integer.MaxValue, Integer.MaxValue), TextFormatFlags.NoPadding).Height

        ' Label has no padding; use a small inset so it lines up visually.
        Dim insetX As Integer = 8
        Dim insetY As Integer = 6

        Dim x As Integer = insetX + (col * charW)
        Dim y As Integer = insetY + (lineIndex * lineH)

        ' Place relative to label within the panel
        txtInline.Left = lblScreen.Left + x
        txtInline.Top = lblScreen.Top + y
        txtInline.Height = lineH
        txtInline.Width = Math.Max(250, pnlScreen.ClientSize.Width - txtInline.Left - 20)
    End Sub

    Private Sub OnFormKeyDown(sender As Object, e As KeyEventArgs)
        If e.KeyCode = Keys.Escape Then
            Me.Close()
            e.Handled = True
            Return
        End If

        If _mode <> InputMode.Menu Then Return

        Select Case e.KeyCode
            Case Keys.D1, Keys.NumPad1
                ShowPromptInline("CUSTOMER NAME?")
                e.Handled = True
                Return

            Case Keys.Q
                Me.Close()
                e.Handled = True
                Return
        End Select
    End Sub

    Private Sub OnInlineKeyDown(sender As Object, e As KeyEventArgs)
        If e.KeyCode = Keys.Escape Then
            RenderMenu()
            e.Handled = True
            Return
        End If

        If e.KeyCode <> Keys.Enter Then Return

        Dim input As String = txtInline.Text.Trim()
        HideInline()

        If _mode = InputMode.ViewOneCustomerPrompt Then
            RunViewOneCustomer(input)
            e.Handled = True
            Return
        End If
    End Sub

    Private Sub RunViewOneCustomer(customerInput As String)
        If String.IsNullOrWhiteSpace(customerInput) Then
            RenderMenu()
            Return
        End If

        Dim reader As New LogBookReader(DataDir)

        Dim pages As New List(Of String)()
        Try
            For Each entry In reader.ReadEntries(_yearTwoDigit)
                If entry Is Nothing Then Continue For

                If entry.Customer IsNot Nothing AndAlso
                   entry.Customer.StartsWith(customerInput, StringComparison.OrdinalIgnoreCase) Then

                    pages.Add(entry.FormatForDosViewer())
                End If
            Next
        Catch ex As Exception
            Using v As New FrmDosPagedViewer()
                v.Text = "LOG BOOK"
                v.SetSinglePage("ERROR READING LOG BOOK:" & vbCrLf & ex.Message)
                v.ShowDialog(Me)
            End Using
            RenderMenu()
            Return
        End Try

        If pages.Count = 0 Then
            Using v As New FrmDosPagedViewer()
                v.Text = "LOG BOOK"
                v.SetSinglePage("END OF LOG BOOK, HIT [ESC] to Exit")
                v.ShowDialog(Me)
            End Using
            RenderMenu()
            Return
        End If

        pages.Add("END OF LOG BOOK, HIT [ESC] to Exit")

        Using viewer As New FrmDosPagedViewer()
            viewer.Text = "LOG BOOK"
            viewer.SetPages(pages)
            viewer.ShowDialog(Me)
        End Using

        RenderMenu()
    End Sub
