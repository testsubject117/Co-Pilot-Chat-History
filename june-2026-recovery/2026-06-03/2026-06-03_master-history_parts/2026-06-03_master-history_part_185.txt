function Get-ExportFilesForDate {
    param(
        [string[]]$Paths,
        [datetime]$Date
    )

    $datePatterns = Get-DatePatterns -Date $Date
    $results = @()

    foreach ($path in $Paths) {
        if (-not (Test-Path $path)) {
            Write-WarnMsg "Scan path not found: $path"
            continue
        }

        Write-Info "Scanning path: $path"

        $files = Get-ChildItem -Path $path -File -ErrorAction SilentlyContinue |
            Where-Object { $_.Extension.ToLower() -in ".md", ".json", ".zip" } |
            Where-Object {
                $baseName = $_.BaseName.ToLower().Replace("-", "_").Replace(" ", "_")
                ($datePatterns | Where-Object { $baseName -like "*$_*" }).Count -gt 0
            }

        if ($files) {
            Write-Info "Found $($files.Count) matching export file(s) in $path"
            $results += $files
        }
    }

    return $results
}

function Expand-ZipFile {
    param(
        [Parameter(Mandatory = $true)]
        [object]$ZipFile,

        [Parameter(Mandatory = $true)]
        [string]$DestinationRoot
    )

    $extractFolderName = [System.IO.Path]::GetFileNameWithoutExtension($ZipFile.Name)
    $extractPath = Join-Path $DestinationRoot $extractFolderName

    if (Test-Path $extractPath) {
        Remove-Item -Path $extractPath -Recurse -Force -ErrorAction SilentlyContinue
    }

    New-Item -ItemType Directory -Path $extractPath -Force | Out-Null
    Expand-Archive -Path $ZipFile.FullName -DestinationPath $extractPath -Force

    Write-Info "Expanded zip: $($ZipFile.FullName)"
    Write-Info "Extracted to: $extractPath"

    return $extractPath
}

function Split-TextFileIntoParts {
    param(
        [Parameter(Mandatory = $true)]
        [string]$SourceFile,

        [Parameter(Mandatory = $true)]
        [string]$OutputRoot,

        [int]$LinesPerPart = 2000
    )

    if (-not (Test-Path $SourceFile)) {
        return $null
    }

    $baseName = [System.IO.Path]::GetFileNameWithoutExtension($SourceFile)
    $partsFolder = Join-Path $OutputRoot ($baseName + "_parts")

    New-Item -ItemType Directory -Path $partsFolder -Force | Out-Null

    Get-ChildItem -Path $partsFolder -File -ErrorAction SilentlyContinue |
        Where-Object { $_.Name -like "${baseName}_part_*.txt" } |
        Remove-Item -Force -ErrorAction SilentlyContinue

    $chunk = New-Object System.Collections.Generic.List[string]
    $partNumber = 1

    Get-Content -Path $SourceFile | ForEach-Object {
        $chunk.Add($_)

        if ($chunk.Count -ge $LinesPerPart) {
            $partPath = Join-Path $partsFolder ("{0}_part_{1:D3}.txt" -f $baseName, $partNumber)
            $chunk | Set-Content -Path $partPath -Encoding UTF8
            $chunk = New-Object System.Collections.Generic.List[string]
            $partNumber++
        }
    }

    if ($chunk.Count -gt 0) {
        $partPath = Join-Path $partsFolder ("{0}_part_{1:D3}.txt" -f $baseName, $partNumber)
        $chunk | Set-Content -Path $partPath -Encoding UTF8
    }

    return $partsFolder
}

function Get-MonthRecoveryFolderName {
    param([datetime]$Date)

    $monthFolder = $Date.ToString("MMM-yyyy").ToLower()

    $monthMap = @{
        'jan' = 'january'
        'feb' = 'february'
        'mar' = 'march'
        'apr' = 'april'
        'may' = 'may'
        'jun' = 'june'
        'jul' = 'july'
        'aug' = 'august'
        'sep' = 'september'
        'oct' = 'october'
        'nov' = 'november'
        'dec' = 'december'
    }

    if ($monthFolder -match '^(jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec)-') {
        $abbr = $matches[1].ToLower()
        return ($monthFolder -replace "^(?i)$abbr", $monthMap[$abbr])
    }

    return $monthFolder
}

function Get-GitOriginUrl {
    param([string]$Path)

    if (-not (Test-Path $Path)) {
        return $null
    }

    if (-not (Test-Path (Join-Path $Path ".git"))) {
        return $null
    }

    Push-Location $Path
    try {
        $origin = git remote get-url origin 2>$null
        if ($LASTEXITCODE -ne 0) {
            return $null
        }

        return ($origin | Select-Object -First 1)
    }
    finally {
        Pop-Location
    }
}

function Assert-RepoIsCorrect {
    param(
        [string]$Path,
        [string]$ExpectedRepo
    )

    if (-not (Test-Path $Path)) {
        throw "Local repo path does not exist: $Path"
    }

    if (-not (Test-Path (Join-Path $Path ".git"))) {
        throw "Local repo path is not a Git repository: $Path"
    }

    $origin = Get-GitOriginUrl -Path $Path
    if (-not $origin) {
        throw "Could not read git origin from repo: $Path"
    }

    if ($origin -notmatch [regex]::Escape($ExpectedRepo)) {
        throw "Repo origin does not match expected GitHub repo '$ExpectedGitHubRepo'. Actual origin: $origin"
    }
}
