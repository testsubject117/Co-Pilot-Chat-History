function Split-TextFileIntoParts {
    param(
        [Parameter(Mandatory = $true)]
        [string]$SourceFile,

        [Parameter(Mandatory = $true)]
        [string]$OutputRoot,

        [int]$LinesPerPart = 2000
    )

    if (-not (Test-Path $SourceFile)) {
        return
    }

    $baseName = [System.IO.Path]::GetFileNameWithoutExtension($SourceFile)
    $partsFolder = Join-Path $OutputRoot ($baseName + "_parts")

    if (-not (Test-Path $partsFolder)) {
        New-Item -ItemType Directory -Path $partsFolder -Force | Out-Null
    }

    # Remove old generated parts so re-runs stay clean
    Get-ChildItem -Path $partsFolder -File -Filter "${baseName}_part_*.txt" -ErrorAction SilentlyContinue |
        Remove-Item -Force -ErrorAction SilentlyContinue

    $chunk = New-Object System.Collections.Generic.List[string]
    $partNumber = 1

    Get-Content -Path $SourceFile | ForEach-Object {
        $chunk.Add($_)

        if ($chunk.Count -ge $LinesPerPart) {
            $partPath = Join-Path $partsFolder ("{0}_part_{1:D3}.txt" -f $baseName, $partNumber)
            $chunk | Set-Content -Path $partPath
            $chunk = New-Object System.Collections.Generic.List[string]
            $partNumber++
        }
    }

    if ($chunk.Count -gt 0) {
        $partPath = Join-Path $partsFolder ("{0}_part_{1:D3}.txt" -f $baseName, $partNumber)
        $chunk | Set-Content -Path $partPath
    }
}

function Get-VersionNumber {
    param([string]$FileName)

    if ($FileName -match '\((\d+)\)(?=\.[^.]+$)') {
        return [int]$matches[1]
    }

    return 0
}

function Get-NormalizedBaseName {
    param([string]$FileName)

    $nameWithoutExtension = [System.IO.Path]::GetFileNameWithoutExtension($FileName)
    $normalized = $nameWithoutExtension -replace '\s*\(\d+\)$', ''
    return $normalized.ToLower()
}

function Get-TodayExportFiles {
    param([string[]]$Paths)

    $now = Get-Date

    $datePatterns = @(
        $now.ToString("ddd_MMM_d_yyyy"),
        $now.ToString("ddd_MMMM_d_yyyy"),
        $now.ToString("yyyy-MM-dd"),
        $now.ToString("MM-dd-yyyy"),
        $now.ToString("M-d-yyyy")
    ) | ForEach-Object {
        $_.ToLower().Replace("-", "_").Replace(" ", "_")
    }

    $results = @()

    foreach ($path in $Paths) {
        if (-not (Test-Path $path)) {
            continue
        }

        $files = Get-ChildItem -Path $path -File -ErrorAction SilentlyContinue |
            Where-Object {
                $_.Extension.ToLower() -in ".md", ".json", ".zip"
            } |
            Where-Object {
                $baseName = $_.BaseName.ToLower().Replace("-", "_").Replace(" ", "_")
                ($datePatterns | Where-Object { $baseName -like "*$_*" }).Count -gt 0
            }

        $results += $files
    }

    return $results
}

function Get-LatestExportGroup {
    param([System.IO.FileInfo[]]$Files)

    if (-not $Files -or $Files.Count -eq 0) {
        return $null
    }

    $ranked = $Files |
        Sort-Object `
            @{ Expression = { Get-VersionNumber $_.Name }; Descending = $true },
            @{ Expression = { $_.LastWriteTime }; Descending = $true }

    $best = $ranked | Select-Object -First 1
    $bestNormalized = Get-NormalizedBaseName $best.Name
    $bestVersion = Get-VersionNumber $best.Name

    $group = $Files | Where-Object {
        (Get-NormalizedBaseName $_.Name) -eq $bestNormalized -and
        (Get-VersionNumber $_.Name) -eq $bestVersion
    }

    return $group
}

function Expand-ZipFile {
    param(
        [System.IO.FileInfo]$ZipFile,
        [string]$DestinationRoot
    )

    $extractFolderName = [System.IO.Path]::GetFileNameWithoutExtension($ZipFile.Name)
    $extractPath = Join-Path $DestinationRoot $extractFolderName

    if (Test-Path $extractPath) {
        Remove-Item -Path $extractPath -Recurse -Force -ErrorAction SilentlyContinue
    }

    New-Item -ItemType Directory -Path $extractPath -Force | Out-Null
    Expand-Archive -Path $ZipFile.FullName -DestinationPath $extractPath -Force

    return $extractPath
}

function Get-MainMarkdownCandidate {
    param(
        [System.IO.FileInfo[]]$MarkdownFiles,
        [string]$ExpectedBaseName
    )

    if (-not $MarkdownFiles -or $MarkdownFiles.Count -eq 0) {
        return $null
    }

    $expected = $ExpectedBaseName.ToLower()

    $ranked = $MarkdownFiles |
        Sort-Object `
            @{ Expression = {
                $name = $_.BaseName.ToLower()
                if ($name -eq $expected) { 100 }
                elseif ($name -like "$expected*") { 90 }
                elseif ($name -match 'checklist|readme|guide|instruction|instructions') { 10 }
                else { 50 }
            }; Descending = $true },
            @{ Expression = { $_.Length }; Descending = $true },
            @{ Expression = { $_.LastWriteTime }; Descending = $true }

    return $ranked | Select-Object -First 1
}

                SourceType    = "ExtractedJson"
                ImportFile    = $jsonInside
                ContainerFile = $zipFile

        }

            SourceType    = "StandaloneJson"
            ImportFile    = $standaloneJson
            ContainerFile = $null
            ExtractPath   = $null
        }
    }

    return $null

function Split-TextFileIntoParts {
    param(
        [Parameter(Mandatory = $true)]
        [string]$SourceFile,

        [Parameter(Mandatory = $true)]
        [string]$OutputRoot,

        [int]$LinesPerPart = 2000
    )

    if (-not (Test-Path $SourceFile)) {
        return
    }

    $baseName = [System.IO.Path]::GetFileNameWithoutExtension($SourceFile)
    $partsFolder = Join-Path $OutputRoot ($baseName + "_parts")

    New-Item -ItemType Directory -Path $partsFolder -Force | Out-Null

    # Clear old generated parts so repeated runs stay clean
    Get-ChildItem -Path $partsFolder -File -Filter "${baseName}_part_*.txt" -ErrorAction SilentlyContinue |
        Remove-Item -Force -ErrorAction SilentlyContinue

    $chunk = New-Object System.Collections.Generic.List[string]
    $partNumber = 1

    Get-Content -Path $SourceFile | ForEach-Object {
        $chunk.Add($_)

        if ($chunk.Count -ge $LinesPerPart) {
            $partPath = Join-Path $partsFolder ("{0}_part_{1:D3}.txt" -f $baseName, $partNumber)
            $chunk | Set-Content -Path $partPath -Encoding UTF8
            $chunk = New-Object System.Collections.Generic.List[string]
            $partNumber++
        }
    }

    if ($chunk.Count -gt 0) {
        $partPath = Join-Path $partsFolder ("{0}_part_{1:D3}.txt" -f $baseName, $partNumber)
        $chunk | Set-Content -Path $partPath -Encoding UTF8
    }
}

function Get-VersionNumber {
    param([string]$FileName)

    if ($FileName -match '\((\d+)\)(?=\.[^.]+$)') {
        return [int]$matches[1]
    }

    return 0
}

function Get-NormalizedBaseName {
    param([string]$FileName)

    $nameWithoutExtension = [System.IO.Path]::GetFileNameWithoutExtension($FileName)
    $normalized = $nameWithoutExtension -replace '\s*\(\d+\)$', ''
    return $normalized.ToLower()
}

function Get-TodayExportFiles {
    param([string[]]$Paths)

    $now = Get-Date

    $datePatterns = @(
        $now.ToString("ddd_MMM_d_yyyy"),
        $now.ToString("ddd_MMMM_d_yyyy"),
        $now.ToString("yyyy-MM-dd"),
        $now.ToString("MM-dd-yyyy"),
        $now.ToString("M-d-yyyy")
    ) | ForEach-Object {
        $_.ToLower().Replace("-", "_").Replace(" ", "_")
    }

    $results = @()

    foreach ($path in $Paths) {
        if (-not (Test-Path $path)) {
            continue
        }

        $files = Get-ChildItem -Path $path -File -ErrorAction SilentlyContinue |
            Where-Object {
                $_.Extension.ToLower() -in ".md", ".json", ".zip"
            } |
            Where-Object {
                $baseName = $_.BaseName.ToLower().Replace("-", "_").Replace(" ", "_")
                ($datePatterns | Where-Object { $baseName -like "*$_*" }).Count -gt 0
            }

        $results += $files
    }

    return $results
}

function Get-LatestExportGroup {
    param([System.IO.FileInfo[]]$Files)

    if (-not $Files -or $Files.Count -eq 0) {
        return $null
    }

    $ranked = $Files |
        Sort-Object `
            @{ Expression = { Get-VersionNumber $_.Name }; Descending = $true },
            @{ Expression = { $_.LastWriteTime }; Descending = $true }

    $best = $ranked | Select-Object -First 1
    $bestNormalized = Get-NormalizedBaseName $best.Name
    $bestVersion = Get-VersionNumber $best.Name

    $group = $Files | Where-Object {
        (Get-NormalizedBaseName $_.Name) -eq $bestNormalized -and
        (Get-VersionNumber $_.Name) -eq $bestVersion
    }

    return $group
}

function Expand-ZipFile {
    param(
        [System.IO.FileInfo]$ZipFile,
        [string]$DestinationRoot
    )

    $extractFolderName = [System.IO.Path]::GetFileNameWithoutExtension($ZipFile.Name)
    $extractPath = Join-Path $DestinationRoot $extractFolderName

    if (Test-Path $extractPath) {
        Remove-Item -Path $extractPath -Recurse -Force -ErrorAction SilentlyContinue
    }

    New-Item -ItemType Directory -Path $extractPath -Force | Out-Null
    Expand-Archive -Path $ZipFile.FullName -DestinationPath $extractPath -Force

    return $extractPath
}

function Get-MainMarkdownCandidate {
    param(
        [System.IO.FileInfo[]]$MarkdownFiles,
        [string]$ExpectedBaseName
    )

    if (-not $MarkdownFiles -or $MarkdownFiles.Count -eq 0) {
        return $null
    }

    $expected = $ExpectedBaseName.ToLower()

    $ranked = $MarkdownFiles |
        Sort-Object `
            @{ Expression = {
                $name = $_.BaseName.ToLower()
                if ($name -eq $expected) { 100 }
                elseif ($name -like "$expected*") { 90 }
                elseif ($name -match 'checklist|readme|guide|instruction|instructions') { 10 }
                else { 50 }
            }; Descending = $true },
            @{ Expression = { $_.Length }; Descending = $true },
            @{ Expression = { $_.LastWriteTime }; Descending = $true }

    return $ranked | Select-Object -First 1
}

function Get-BestImportSourceFromGroup {

                SourceType    = "ExtractedJson"
                ImportFile    = $jsonInside
