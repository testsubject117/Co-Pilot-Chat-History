Imports System
Imports System.IO
Imports System.Text

Module AreaCodeFileConverter

    Sub Main()
        Dim inputPath As String = "\\invoice\mainmenu\data\phone\areacodes.csv"
        Dim backupPath As String = "\\invoice\mainmenu\data\phone\areacodes_backup.csv"
        Dim outputPath As String = "\\invoice\mainmenu\data\phone\areacodes.csv"

        If Not File.Exists(inputPath) Then
            Console.WriteLine("Input file not found: " & inputPath)
            Return
        End If

        File.Copy(inputPath, backupPath, True)

        Dim lines = File.ReadAllLines(inputPath)
        Dim converted As New List(Of String)

        For Each line In lines
            If String.IsNullOrWhiteSpace(line) Then
                converted.Add(line)
                Continue For
            End If

            Dim parts = line.Split(New Char() {","c}, 4)

            If parts.Length = 4 Then
                converted.Add(parts(0).Trim() & "|" &
                              parts(1).Trim() & "|" &
                              parts(2).Trim() & "|" &
                              parts(3).Trim())
            Else
                converted.Add(line)
            End If
        Next

        Console.WriteLine("Backup created: " & backupPath)
        Console.WriteLine("Updated file: " & outputPath)
    End Sub

End Module
