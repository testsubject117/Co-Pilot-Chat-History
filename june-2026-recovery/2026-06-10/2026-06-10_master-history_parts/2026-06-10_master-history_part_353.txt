Render the prompt on the screen, and use a dedicated input TextBox that:
- is placed on the same line as the prompt (not bottom)
- is guaranteed visible (weâ€™ll give it a visible border while debugging)
- does not rely on row/col math inside a Label/TextBox

We can do that by using a **TableLayoutPanel** with:
- top: a read-only â€œscreenâ€ TextBox
- bottom: a single prompt row (but we position it so it visually sits right under the prompt line area, not at the very bottom)
However you explicitly want it next to the prompt line in the screen. The only robust way to do â€œnext to promptâ€ without fragile text measurement is to render the whole screen ourselves.

### Robust terminal approach (works, minimal code):
Use a custom control that draws text with `TextRenderer.DrawText` and we draw the caret position ourselves, then use a hidden TextBox for input capture. Thatâ€™s more code.

Given the time, hereâ€™s a pragmatic compromise that matches DOS *behavior* and is visually close:

- Prompt appears **immediately under the menu** (where it already is).
- Input box appears on the **same line**, directly after `CUSTOMER NAME?`.
- It is *not at the bottom*; it is near the prompt line.
- Achieved by placing the prompt row directly under the menu area (not bottom dock).

We can do this by:
- Keep the screen as a **read-only TextBox** docked fill.
- Add a prompt overlay **Panel** positioned at the Y coordinate of the prompt line (computed by counting lines, which is stable in a monospaced TextBox), and in that panel put label + input textbox.
- This avoids overlaying directly on the TextBox itself and avoids label measurement drift.

    Private Sub RenderMenu(Optional promptLineVisible As Boolean = False)
        _mode = InputMode.Menu

        lblPrompt.Text = ""

        sb.AppendLine("(2) View all customers")
        sb.AppendLine("(3) Look up an invoice")
        sb.AppendLine("(4) Look up a P.O. Number")
        sb.AppendLine("(5) Look up the last date of business for a customer")

    Private Sub PositionPromptOverlay()
        ' Place overlay at the Y-position where "CUSTOMER NAME?" line is in the screen.
        ' Since txtScreen is monospaced and not scrolling, line-height math is stable.
        Dim allLines As String() = txtScreen.Text.Split(New String() {vbCrLf}, StringSplitOptions.None)

        Dim promptLineIndex As Integer = -1
        For i As Integer = allLines.Length - 1 To 0 Step -1
            If allLines(i).Trim().Equals("CUSTOMER NAME?", StringComparison.OrdinalIgnoreCase) Then
                promptLineIndex = i
                Exit For
            End If
        Next

        Dim lineH As Integer = TextRenderer.MeasureText("W", txtScreen.Font).Height

        ' A small inset to match textbox internal padding
        Dim insetX As Integer = txtScreen.Left + 2
        Dim insetY As Integer = txtScreen.Top + 2

        If promptLineIndex < 0 Then
            promptLineIndex = allLines.Length - 1
        End If

        pnlPrompt.Left = insetX
        pnlPrompt.Top = insetY + (promptLineIndex * lineH)
        pnlPrompt.Width = txtScreen.ClientSize.Width - 4
        pnlPrompt.Height = lineH + 4

        lblPrompt.Left = 0
        lblPrompt.Top = 2

        Dim promptW As Integer = TextRenderer.MeasureText(lblPrompt.Text, lblPrompt.Font).Width
        txtInput.Left = promptW
        txtInput.Top = 0
        txtInput.Height = pnlPrompt.Height - 2
        txtInput.Width = Math.Max(250, pnlPrompt.Width - txtInput.Left - 10)
    End Sub

    Private Sub OnFormKeyDown(sender As Object, e As KeyEventArgs)
        If _mode <> InputMode.Menu Then Return

        Select Case e.KeyCode
            Case Keys.Escape
                Me.Close()
                e.Handled = True

            Case Keys.D1, Keys.NumPad1
                BeginCustomerPrompt()
                e.Handled = True

            Case Keys.Q
                Me.Close()
                e.Handled = True
        End Select
    End Sub

    Private Sub OnInputKeyDown(sender As Object, e As KeyEventArgs)

            RenderMenu()
            pnlPrompt.Visible = False
            _mode = InputMode.Menu
            e.Handled = True
            Return
        End If

        If e.KeyCode <> Keys.Enter Then Return

        Dim input As String = txtInput.Text.Trim()
        pnlPrompt.Visible = False

        RunViewOneCustomer(input)
        e.Handled = True
    End Sub

    Private Sub RunViewOneCustomer(customerInput As String)
        If String.IsNullOrWhiteSpace(customerInput) Then
            RenderMenu()
            Return
        End If

        Dim reader As New LogBookReader(DataDir)

        Dim pages As New List(Of String)()
        Try
            For Each entry In reader.ReadEntries(_yearTwoDigit)
                If entry Is Nothing Then Continue For

                If entry.Customer IsNot Nothing AndAlso
                   entry.Customer.StartsWith(customerInput, StringComparison.OrdinalIgnoreCase) Then

                    pages.Add(entry.FormatForDosViewer())
                End If
            Next
        Catch ex As Exception
            Using v As New FrmDosPagedViewer()
                v.Text = "LOG BOOK"
                v.SetSinglePage("ERROR READING LOG BOOK:" & vbCrLf & ex.Message)
                v.ShowDialog(Me)
            End Using
            RenderMenu()
            Return
        End Try

        If pages.Count = 0 Then
            Using v As New FrmDosPagedViewer()

                v.ShowDialog(Me)
            End Using

        Using viewer As New FrmDosPagedViewer()
            viewer.Text = "LOG BOOK"
            viewer.SetPages(pages)
            viewer.ShowDialog(Me)
        End Using

        RenderMenu()
    End Sub
End Class

        _mode = InputMode.ViewOneCustomerPrompt

        ' Show the prompt line in the screen
        RenderMenu(promptLineVisible:=True)

        lblPrompt.Text = "CUSTOMER NAME? "

        txtInput.BringToFront()

    Private Sub PositionPromptOverlay()
        ' Place overlay at the Y-position where "CUSTOMER NAME?" line is in the screen.
        ' Since txtScreen is monospaced and not scrolling, line-height math is stable.
        Dim allLines As String() = txtScreen.Text.Split(New String() {vbCrLf}, StringSplitOptions.None)

        Dim promptLineIndex As Integer = -1
        For i As Integer = allLines.Length - 1 To 0 Step -1
            If allLines(i).Trim().Equals("CUSTOMER NAME?", StringComparison.OrdinalIgnoreCase) Then
                promptLineIndex = i
                Exit For
            End If
        Next

        Dim lineH As Integer = TextRenderer.MeasureText("W", txtScreen.Font).Height

        ' A small inset to match textbox internal padding
        Dim insetX As Integer = txtScreen.Left + 2
        Dim insetY As Integer = txtScreen.Top + 2

        If promptLineIndex < 0 Then
            promptLineIndex = allLines.Length - 1
        End If

        pnlPrompt.Left = insetX
        pnlPrompt.Top = insetY + (promptLineIndex * lineH)
        pnlPrompt.Width = txtScreen.ClientSize.Width - 4
        pnlPrompt.Height = lineH + 4

        lblPrompt.Left = 0
        lblPrompt.Top = 2

        Dim promptW As Integer = TextRenderer.MeasureText(lblPrompt.Text, lblPrompt.Font).Width
        txtInput.Left = promptW
        txtInput.Top = 0
        txtInput.Height = pnlPrompt.Height - 2
        txtInput.Width = Math.Max(250, pnlPrompt.Width - txtInput.Left - 10)
    End Sub

    Private Sub OnFormKeyDown(sender As Object, e As KeyEventArgs)
        If _mode <> InputMode.Menu Then Return

        Select Case e.KeyCode
            Case Keys.Escape
                Me.Close()
                e.Handled = True

            Case Keys.D1, Keys.NumPad1
                BeginCustomerPrompt()
                e.Handled = True

            Case Keys.Q
                Me.Close()
                e.Handled = True
        End Select
    End Sub

    Private Sub OnInputKeyDown(sender As Object, e As KeyEventArgs)
        If e.KeyCode = Keys.Escape Then
            RenderMenu()
            pnlPrompt.Visible = False
            _mode = InputMode.Menu
            e.Handled = True
            Return
        End If

        If e.KeyCode <> Keys.Enter Then Return

        Dim input As String = txtInput.Text.Trim()
        pnlPrompt.Visible = False

        RunViewOneCustomer(input)
        e.Handled = True
    End Sub

    Private Sub RunViewOneCustomer(customerInput As String)
        If String.IsNullOrWhiteSpace(customerInput) Then
            RenderMenu()
            Return
        End If

        Dim reader As New LogBookReader(DataDir)

        Dim pages As New List(Of String)()
        Try
            For Each entry In reader.ReadEntries(_yearTwoDigit)
                If entry Is Nothing Then Continue For

                If entry.Customer IsNot Nothing AndAlso
                   entry.Customer.StartsWith(customerInput, StringComparison.OrdinalIgnoreCase) Then

                    pages.Add(entry.FormatForDosViewer())
                End If
            Next
        Catch ex As Exception
            Using v As New FrmDosPagedViewer()
                v.Text = "LOG BOOK"
                v.SetSinglePage("ERROR READING LOG BOOK:" & vbCrLf & ex.Message)
                v.ShowDialog(Me)
            End Using
            RenderMenu()
            Return
        End Try

        If pages.Count = 0 Then
            Using v As New FrmDosPagedViewer()
                v.Text = "LOG BOOK"
                v.SetSinglePage("END OF LOG BOOK, HIT [ESC] to Exit")
                v.ShowDialog(Me)
            End Using
            RenderMenu()
            Return
        End If

        pages.Add("END OF LOG BOOK, HIT [ESC] to Exit")

        Using viewer As New FrmDosPagedViewer()
            viewer.Text = "LOG BOOK"
            viewer.SetPages(pages)
            viewer.ShowDialog(Me)
        End Using

        RenderMenu()
    End Sub
End Class
