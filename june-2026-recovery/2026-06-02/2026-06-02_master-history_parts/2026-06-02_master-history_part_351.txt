    Public Shared Function NormalizeTwoDigitYear(input As String, defaultYear As Integer) As Integer
        Dim t As String = If(input, "").Trim()
        If t.Length = 0 Then
            Return defaultYear Mod 100
        End If

        Dim n As Integer
        If Not Integer.TryParse(t, n) Then
            Return defaultYear Mod 100
        End If

        ' Accept 2-digit or 4-digit.
        If n >= 0 AndAlso n <= 99 Then Return n
        If n >= 1900 AndAlso n <= 2999 Then Return n Mod 100

        Return defaultYear Mod 100
    End Function

    Public Function GetLogBookPath(twoDigitYear As Integer) As String
        Dim yy As String = twoDigitYear.ToString("00")
        Return Path.Combine(_baseDir, $"LOGBOOK.{yy}")
    End Function

    Public Iterator Function ReadEntries(twoDigitYear As Integer) As IEnumerable(Of LogBookEntry)
        Dim path As String = GetLogBookPath(twoDigitYear)
        If Not File.Exists(path) Then
            Throw New FileNotFoundException("Log book file not found.", path)
        End If

        Using fs As New FileStream(path, FileMode.Open, FileAccess.Read, FileShare.ReadWrite)
            Using sr As New StreamReader(fs)
                Dim buffer As New List(Of String)(capacity:=12)

                While True
                    Dim line As String = sr.ReadLine()
                    If line Is Nothing Then Exit While

                    buffer.Add(line)

                    If buffer.Count = 12 Then
                        Yield ParseEntry(buffer)
                        buffer.Clear()
                    End If
                End While

                ' If there is a partial entry at EOF, ignore it (defensive).
            End Using
        End Using
    End Function

    Private Shared Function ParseEntry(lines As List(Of String)) As LogBookEntry
        ' Expected 12 lines per entry.
        Dim e As New LogBookEntry()

        Dim dateRaw As String = Unquote(lines(0))
        Dim cust As String = Unquote(lines(1))
        Dim part As String = Unquote(lines(2))

        Dim invRaw As String = lines(3).Trim()
        Dim po As String = Unquote(lines(4))
        Dim spec As String = Unquote(lines(5))
        Dim qtyAcc As String = Unquote(lines(6))
        Dim qtyRej As String = Unquote(lines(7))
        Dim mat As String = Unquote(lines(8))
        Dim ht As String = Unquote(lines(9))
        Dim rsn As String = Unquote(lines(10))
        Dim st As String = Unquote(lines(11))

        e.Customer = cust.Trim()
        e.Part = part.Trim()
        e.PONumber = po.Trim()
        e.Spec = spec.Trim()
        e.QtyAcceptedOrReceived = qtyAcc.Trim()
        e.QtyRejected = qtyRej.Trim()
        e.Material = mat.Trim()
        e.HeatTreat = ht.Trim()
        e.RejectReason = rsn.Trim()
        e.Status = st.Trim()

        Dim inv As Integer
        If Integer.TryParse(invRaw, inv) Then
            e.InvoiceNumber = inv
        End If

        ' Dates appear like 01-02-2014 in your sample.
        Dim d As Date
        If Date.TryParseExact(dateRaw.Trim(),
                              "MM-dd-yyyy",
                              CultureInfo.InvariantCulture,
                              DateTimeStyles.None,
                              d) Then
            e.EntryDate = d
        End If

        Return e
    End Function

    Private Shared Function Unquote(s As String) As String
        If s Is Nothing Then Return ""
        Dim t As String = s.Trim()

        If t.Length >= 2 AndAlso t.StartsWith("""", StringComparison.Ordinal) AndAlso t.EndsWith("""", StringComparison.Ordinal) Then
            t = t.Substring(1, t.Length - 2)
        End If

        Return t
    End Function

End Class

## File 2 â€” `LogBookReader.vb` (Class)
```vbnet name=LogBookReader.vb
Option Strict On
Option Explicit On

Imports System
Imports System.Collections.Generic
Imports System.Globalization
Imports System.IO

Public Class LogBookReader

    Private ReadOnly _baseDir As String

    Public Sub New(baseDir As String)
        _baseDir = baseDir
    End Sub

    Public Function GetLogBookPath(twoDigitYear As Integer) As String
        Dim yy As String = twoDigitYear.ToString("00")
        Return Path.Combine(_baseDir, $"LOGBOOK.{yy}")
    End Function

    Public Iterator Function ReadEntries(twoDigitYear As Integer) As IEnumerable(Of LogBookEntry)
        Dim path As String = GetLogBookPath(twoDigitYear)
        If Not File.Exists(path) Then
            Throw New FileNotFoundException("Log book file not found.", path)
        End If

        Using fs As New FileStream(path, FileMode.Open, FileAccess.Read, FileShare.ReadWrite)
            Using sr As New StreamReader(fs)
                Dim buffer As New List(Of String)(capacity:=12)

                While True
                    Dim line As String = sr.ReadLine()
                    If line Is Nothing Then Exit While

                    buffer.Add(line)

                    If buffer.Count = 12 Then
                        Yield ParseEntry(buffer)
                        buffer.Clear()
                    End If
                End While

                ' Defensive: ignore partial record at EOF if present.
            End Using
        End Using
    End Function

    Private Shared Function ParseEntry(lines As List(Of String)) As LogBookEntry
        ' Expected 12 lines per entry.
        Dim e As New LogBookEntry()

        Dim dateRaw As String = Unquote(lines(0))
        Dim cust As String = Unquote(lines(1))
        Dim part As String = Unquote(lines(2))

        Dim invRaw As String = lines(3).Trim()
        Dim po As String = Unquote(lines(4))
        Dim spec As String = Unquote(lines(5))
        Dim qtyAcc As String = Unquote(lines(6))
        Dim qtyRej As String = Unquote(lines(7))
        Dim mat As String = Unquote(lines(8))
        Dim ht As String = Unquote(lines(9))
        Dim rsn As String = Unquote(lines(10))
        Dim st As String = Unquote(lines(11))

        e.Customer = cust.Trim()
        e.Part = part.Trim()
        e.PONumber = po.Trim()
        e.Spec = spec.Trim()
        e.QtyAcceptedOrReceived = qtyAcc.Trim()
        e.QtyRejected = qtyRej.Trim()
        e.Material = mat.Trim()
        e.HeatTreat = ht.Trim()
        e.RejectReason = rsn.Trim()
        e.Status = st.Trim()

        Dim inv As Integer
        If Integer.TryParse(invRaw, inv) Then
            e.InvoiceNumber = inv
        End If

        ' Date format from your sample: 01-02-2014
        Dim d As Date
        If Date.TryParseExact(dateRaw.Trim(),
                              "MM-dd-yyyy",
                              CultureInfo.InvariantCulture,
                              DateTimeStyles.None,
                              d) Then
            e.EntryDate = d
        End If

        Return e
    End Function

    Private Shared Function Unquote(s As String) As String
        If s Is Nothing Then Return ""
        Dim t As String = s.Trim()

        If t.Length >= 2 AndAlso t.StartsWith("""", StringComparison.Ordinal) AndAlso t.EndsWith("""", StringComparison.Ordinal) Then
            t = t.Substring(1, t.Length - 2)
        End If

        Return t
    End Function

End Class
```

```
    Private Sub EnsurePromptUi()
        If pnlPromptContainer IsNot Nothing Then Return

        ' Container provides the gray strip visible both above and below the prompt box.
        pnlPromptContainer = New Panel() With {
            .Visible = False,
            .Height = 28 + (PromptLiftPx * 2),
            .Dock = DockStyle.Bottom,
            .BackColor = Color.FromArgb(45, 45, 45),
            .Padding = New Padding(0)
        }

        ' Inner prompt is the black box with white text.
        pnlPromptHost = New Panel() With {
            .Visible = True,
            .Height = 28,
            .BackColor = Color.Black,
            .Padding = New Padding(10, 3, 10, 3)
        }

        lblPrompt = New Label() With {
            .Dock = DockStyle.Fill,
            .BackColor = Color.Black,
            .ForeColor = Color.White,
            .TextAlign = ContentAlignment.MiddleLeft,
            .Font = New Font("Consolas", 12.0F, FontStyle.Bold, GraphicsUnit.Point),
            .AutoEllipsis = True
        }

        pnlPromptHost.Controls.Add(lblPrompt)
        pnlPromptContainer.Controls.Add(pnlPromptHost)
        Me.Controls.Add(pnlPromptContainer)

        pnlPromptContainer.BringToFront()
        LayoutPromptWithinContainer()

        ' Draw a thicker border around the prompt box
        AddHandler pnlPromptHost.Paint, AddressOf pnlPromptHost_Paint
    End Sub

    Private Sub LayoutPromptWithinContainer()
        If pnlPromptContainer Is Nothing OrElse pnlPromptHost Is Nothing Then Return

        pnlPromptHost.Left = 0
        pnlPromptHost.Width = pnlPromptContainer.ClientSize.Width
        pnlPromptHost.Top = PromptLiftPx
    End Sub

    Private Sub pnlPromptHost_Paint(sender As Object, e As PaintEventArgs)
        Dim p As Panel = DirectCast(sender, Panel)

        Using pen As New Pen(_promptBorderColor, CSng(PromptBorderPx))
            ' Keep border inside bounds so it is fully visible; account for pen thickness
            Dim inset As Integer = CInt(Math.Ceiling(PromptBorderPx / 2.0R))
            Dim r As Rectangle = p.ClientRectangle
            r.X += inset
            r.Y += inset
            r.Width -= (inset * 2) + 1
            r.Height -= (inset * 2) + 1

            If r.Width > 0 AndAlso r.Height > 0 Then
                e.Graphics.DrawRectangle(pen, r)
            End If
        End Using
    End Sub

    Private Sub ShowBottomPrompt(text As String)
        EnsurePromptUi()

        lblPrompt.Text = text
```
